 63class Agent(LevelAgent):
 64    """Multi-Agent Proximal Policy Optimization (MAPPO) agent implementation.
 65
 66    This class implements a MAPPO agent that can train and act in multi-agent environments.
 67    It inherits from LevelAgent (so it can be used in a hierarchical agent, but can also be used as a standalone algorithm)
 68    and implements policy optimization using the MAPPO algorithm.
 69    It is mainly a wrapper around the MAPPO implementation in `external_algo/mappo.py`.
 70
 71    Args:
 72        observation_space (GymDict): The observation space for each agent in the environment
 73        action_space (GymDict): The action space for each agent in the environment
 74        device: The device (CPU/GPU) to run computations on
 75        communication_space (GymDict, optional): Communication space if agents can communicate. Defaults to None.
 76        name (str, optional): Name identifier for the agent. Defaults to "mappo".
 77        args (Args, optional): Configuration arguments for the agent. Defaults to None.
 78        torch_compile (bool, optional): Whether to compile model with torch.compile(). Defaults to False.
 79
 80    Attributes:
 81        observation_space (GymDict): Stored observation space
 82        action_space (GymDict): Stored action space
 83        communication_space (GymDict): Stored communication space
 84        args (Args): Configuration arguments
 85        device: Device for computations
 86        name (str): Agent identifier
 87        mappo_args (SimpleNamespace): Arguments specific to MAPPO algorithm
 88        mappo (MAPPO_MPE): MAPPO policy network
 89        replay_buffer (ReplayBuffer): Buffer for storing experience
 90        trained (bool): Whether agent has been trained
 91
 92    Methods:
 93        get_spaces(): Processes observation and action spaces
 94        seed(seed): Sets random seeds
 95        act(observation): Selects actions in evaluation mode
 96        act_train(observation, global_step): Selects actions in training mode
 97        store(state, action, reward, done): Stores transitions in replay buffer
 98        update_step(global_step, writer): Performs training updates
 99        train(env, log_path, run_name): Trains the agent
100        save_agent(save_path, name): Saves trained model
101        load_agent(load_path, name): Loads trained model
102
103    Notes:

180    def get_spaces(self):
181        """
182        Computes and sets observation and action space dimensions and types from environment spaces.
183
184        This method processes the observation and action spaces for all agents to determine:
185        - Total observation size across all agents
186        - Individual agent observation size
187        - Number of agents in environment
188        - Action space dimensionality
189        - Whether actions are continuous or discrete
190
191        Raises:
192            ValueError: If action space is neither Discrete nor Box type from gymnasium
193
194        Note:
195            Expects self.observation_space and self.action_space to be set before calling.
196            Sets the following instance attributes:
197                - self.obs_size: Combined size of observations across all agents
198                - self.obs_size_single: Size of single agent observation
199                - self.num_env_agents: Total number of agents
200                - self.actions_dim: Dimension of action space
201                - self.continuous_actions: Boolean indicating if actions are continuous
202        """
203        self.obs_size = 0
204        self.num_env_agents = 0
205        for agent in self.observation_space:
206            obs_space = self.observation_space[agent]
207            self.obs_size += obs_space.shape[0]  # type: ignore
208            self.obs_size_single = obs_space.shape[0]  # type: ignore
209            self.num_env_agents += 1
210
211        self.actions_dim = 0
212        self.continuous_actions = False
213        for agent in self.action_space:
214            act_space = self.action_space[agent]
215            if isinstance(act_space, Box):
216                self.continuous_actions = True
217                self.actions_dim = max(self.actions_dim, act_space.shape[0])
218            elif isinstance(act_space, Discrete):
219                self.actions_dim = max(self.actions_dim, act_space.n)  # type: ignore
220            else:
221                raise ValueError("Only Discrete or Box gymnasium spaces are supported")

241    def act(self, observation: dict) -> dict:
242        """
243        Takes observations from agents and returns actions using MAPPO policy.
244
245        Args:
246            observation (dict): Dictionary mapping agent IDs to their observations.
247
248        Returns:
249            dict: Dictionary mapping agent IDs to their selected actions.
250
251        Example:
252            >>> agent = MAPPOAgent()
253            >>> obs = {'agent1': [1,2,3], 'agent2': [4,5,6]}
254            >>> actions = agent.act(obs)
255            >>> print(actions)
256            {'agent1': 0, 'agent2': 1}
257        """
258        obs_n = np.array(list(observation.values()))
259        actions, _ = self.mappo.choose_action(obs_n, evaluate=True)
260        return {agent: action for agent, action in zip(observation.keys(), actions)}

262    def act_train(self, observation: Dict[str, np.ndarray], global_step: int) -> dict:
263        """
264        Train-time action selection for MAPPO agents.
265
266        Takes observations from all agents and returns actions using the MAPPO policy.
267        Stores current log probabilities for later use in training.
268
269        Args:
270            observation (Dict[str, np.ndarray]): Dictionary mapping agent IDs to their observations
271            global_step (int): Current training step count
272
273        Returns:
274            dict: Dictionary mapping agent IDs to their selected actions
275
276        Note:
277            This method runs in training mode (evaluate=False) which enables exploration
278        """
279        obs_n = np.array(list(observation.values()))
280        actions, logprobs = self.mappo.choose_action(obs_n, evaluate=False)
281        self.current_logprobs = logprobs
282        return {agent: action for agent, action in zip(observation.keys(), actions)}

284    def store(
285        self,
286        state: dict,
287        action: dict | None = None,
288        reward: float | None = None,
289        done: bool | None = None,
290    ):
291        """Store a transition in the replay buffer.
292
293        This method stores transitions or final values in a replay buffer for training.
294        It handles both regular transitions during episodes
295        and final values at episode termination.
296
297        Args:
298            state (dict): Current state/observation for each agent
299            action (dict, optional): Actions taken by each agent. None if storing final values
300            reward (float, optional): Reward received. None if storing final values
301            done (bool, optional): Whether episode terminated. None if storing final values
302
303        Notes:
304            - If action is None, only stores the final value for the current state
305            - For regular transitions, stores the full transition tuple including:
306                - observations
307                - flattened state
308                - values
309                - actions
310                - action log probabilities
311                - rewards
312                - done flags
313            - Maintains an internal step counter for episode tracking
314        """
315        # Store transition in replay buffer
316        if not hasattr(self, "step_count"):
317            self.step_count = 0
318
319        obs_n = np.array(list(state.values()))
320        s = obs_n.flatten()
321        if action is None:
322            v_n = self.mappo.get_value(s)
323            self.replay_buffer.store_last_value(episode_step=self.step_count, v_n=v_n)  # type: ignore
324            self.step_count = 0
325        else:
326            a_n = np.array(list(action.values()))
327            r_n = np.array([reward for _ in state.keys()])
328            done_n = np.array([done for _ in state.keys()])
329            v_n = self.mappo.get_value(s)
330
331            self.replay_buffer.store_transition(
332                episode_step=self.step_count,
333                obs_n=obs_n,
334                s=s,
335                v_n=v_n,
336                a_n=a_n,
337                a_logprob_n=self.current_logprobs,
338                r_n=r_n,
339                done_n=done_n,
340            )
341            self.step_count += 1

344        """Update step for training the MAPPO agent.
345
346        This method is called during the environment step when training is enabled. If the replay buffer
347        contains enough episodes (>= batch_size), it triggers the MAPPO training process and resets the buffer.
348
349        Args:
350            global_step (int): The current global step count in training
351            writer (SummaryWriter | None): TensorBoard SummaryWriter instance for logging. Can be None.
352
353        Returns:
354            None
355
356        Note:
357            - Only triggers training when replay buffer has collected enough episodes
358            - Resets replay buffer and step count after training

361        if self.replay_buffer.episode_num >= self.args.batch_size:  # type: ignore
362            self.mappo.train(self.replay_buffer, global_step, writer=writer)
363            self.replay_buffer.reset_buffer()
364            self.step_count = 0

366    def train(
367        self,
368        env: ParallelEnv,
369        log_path: Path | str | None = None,
370        run_name: str | None = None,
371    ):
372        """Train the MAPPO agent on a parallel environment.
373
374        This method implements the training loop for Multi-Agent Proximal Policy Optimization (MAPPO).
375        It handles environment interactions, data collection, and agent updates.
376
377        Args:
378            env (ParallelEnv): A parallel environment instance following the PettingZoo parallel API
379            log_path (Path | str | None, optional): Path where to save training logs and checkpoints.
380                Defaults to None, which uses "runs" directory.
381            run_name (str | None, optional): Name for the training run.
382                Defaults to None, which generates a name using experiment name, seed and timestamp.
383
384        The method:
385        - Sets up logging directories and tensorboard writer
386        - Creates a replay buffer for storing transitions
387        - Runs episodes collecting experience until total_timesteps is reached
388        - Periodically trains the MAPPO agent using collected experience
389        - Logs training metrics and saves episode traces
390
391        Training continues until the total number of environment steps reaches args.total_timesteps.
392        Episode traces and tensorboard logs are saved in the specified log directory.
393
394        Returns:
395            None: The method updates the agent in-place and sets self.trained to True upon completion.
396        """
397        if run_name is None:
398            run_name = f"{self.args.exp_name}__{self.args.seed}__{int(time.time())}"  # type: ignore
399
400        if log_path is None:
401            log_path = Path("runs")
402        else:
403            log_path = Path(log_path)
404
405        if not (log_path / run_name).exists():
406            os.makedirs(log_path / run_name)
407
408        args_dict = self.args.to_dict()  # type: ignore
409        with open(log_path / run_name / "params.json", "w") as f:
410            json.dump(args_dict, f, indent=4)
411
412        save_path = log_path / run_name / "training"
413
414        writer = SummaryWriter(save_path / "tboard")
415        writer.add_text(
416            "hyperparameters",
417            "|param|value|\n|-|-|\n%s"
418            % (
419                "\n".join(
420                    [f"|{key}|{value}|" for key, value in vars(self.args).items()]
421                )
422            ),
423        )
424        total_steps = 0
425
426        # Create replay buffer
427        replay_buffer = ReplayBuffer(self.mappo_args)
428        self.agent_names = []
429
430        if self.args.save_all_trace:  # type: ignore
431            trace = Trace()
432        else:
433            trace = RewardTrace()
434
435        pbar = tqdm(
436            total=self.args.total_timesteps,  # type: ignore
437            desc="Training step:",
438        )
439        episode_counter = -1
440        while total_steps < self.args.total_timesteps:  # type: ignore
441            # Run episode
442            observation, _ = env.reset()
443            if len(self.agent_names) == 0:
444                self.agent_names = list(observation.keys())
445            episode_reward = 0
446
447            obs_n = np.array(list(observation.values()))
448            episode_counter += 1
449            for episode_step in range(self.args.episode_length):  # type: ignore
450                # Get actions and values
451                a_n, a_logprob_n = self.mappo.choose_action(obs_n, evaluate=False)
452                s = obs_n.flatten()
453                v_n = self.mappo.get_value(s)
454
455                # Convert actions to dict
456                actions = {agent: a_n[i] for i, agent in enumerate(self.agent_names)}
457
458                # Environment step
459                next_obs, rewards, terminated, truncated, infos = env.step(actions)
460
461                # Process rewards and dones
462                r_n = np.array([rewards[agent] for agent in self.agent_names])
463                done_n = np.array(
464                    [
465                        terminated[agent] or truncated[agent]
466                        for agent in self.agent_names
467                    ]
468                )
469
470                # Store transition
471                replay_buffer.store_transition(
472                    episode_step, obs_n, s, v_n, a_n, a_logprob_n, r_n, done_n
473                )
474
475                trace.add(
476                    actions=actions,
477                    observations=observation,
478                    rewards=rewards,
479                    terminations=terminated,
480                    truncations=truncated,
481                    infos=infos,
482                    episode=episode_counter,
483                )
484
485                observation = next_obs
486                obs_n = np.array(list(observation.values()))
487                episode_reward += np.mean(r_n)
488                total_steps += 1
489                pbar.update(1)
490
491                if all(done_n):
492                    break
493
494            # Store last value
495            s = obs_n.flatten()
496            v_n = self.mappo.get_value(s)
497            replay_buffer.store_last_value(episode_step + 1, v_n)  # type: ignore
498
499            if hasattr(trace, "add_final_obs"):
500                trace.add_final_obs(next_obs)  # type: ignore
501            trace.save_trace(save_path=save_path, episode=str(episode_counter))
502            trace.empty()
503
504            # Train when buffer is ready
505            if replay_buffer.episode_num == self.args.batch_size:  # type: ignore
506                self.mappo.train(replay_buffer, total_steps)
507                replay_buffer.reset_buffer()
508
509            writer.add_scalar("charts/episodic_reward", episode_reward, total_steps)
510
511        writer.close()
512        self.trained = True

514    def _prepare_batch(self, episodes: List[List[Dict]]) -> dict:
515        """Prepares a batch of experiences for training from collected episodes.
516
517        This method processes a list of episodes into a batch dictionary containing observations,
518        states, actions, rewards, done flags and value estimates for all agents. The final batch
519        is converted to PyTorch tensors.
520
521        Args:
522            episodes (List[List[Dict]]): List of episodes, where each episode is a list of
523                timesteps containing dictionaries with experience data.
524
525        Returns:
526            dict: A dictionary containing the processed batch data with the following keys:
527                - obs_n: Observations for all agents
528                - s: Global states
529                - a_n: Actions taken by all agents
530                - r_n: Rewards received by all agents
531                - done_n: Done flags for all agents (all zeros since per-agent dones aren't tracked)
532                - v_n: Value estimates for each state including a zero terminal value
533
534        Note:
535            The returned tensors have shape (num_episodes, episode_length, ...) where the
536            remaining dimensions depend on the specific key.
537        """
538        batch = {"obs_n": [], "s": [], "a_n": [], "r_n": [], "done_n": [], "v_n": []}
539
540        for episode in episodes:
541            episode_obs_n = []
542            episode_states = []
543            episode_actions = []
544            episode_rewards = []
545            episode_dones = []
546            episode_values = []
547
548            for step in episode:
549                episode_obs_n.append(step["obs_n"])
550                episode_states.append(step["state"])
551                episode_actions.append(step["actions"])
552                episode_rewards.append(step["rewards"])
553                episode_dones.append(
554                    np.zeros(self.num_env_agents)
555                )  # We don't track per-agent dones
556                episode_values.append(self.mappo.get_value(step["state"]))
557
558            # Add final value estimate
559            episode_values.append(
560                np.zeros_like(episode_values[0])
561            )  # Zero for terminal state
562
563            batch["obs_n"].append(episode_obs_n)
564            batch["s"].append(episode_states)
565            batch["a_n"].append(episode_actions)
566            batch["r_n"].append(episode_rewards)
567            batch["done_n"].append(episode_dones)
568            batch["v_n"].append(episode_values)
569
570        # Convert to torch tensors
571        for key in batch:
572            batch[key] = torch.FloatTensor(np.array(batch[key]))  # type: ignore
573
574        return batch
