 60class ContinuousActionAgent(nn.Module):
 61    def __init__(
 62        self,
 63        obs_size: int,
 64        actions_size: int,
 65        device: torch.device,
 66        torch_compile: bool = True,
 67    ):
 68        super().__init__()
 69        self.device = device
 70        self.critic = nn.Sequential(
 71            layer_init(nn.Linear(obs_size, 64)),
 72            nn.Tanh(),
 73            layer_init(nn.Linear(64, 64)),
 74            nn.Tanh(),
 75            layer_init(nn.Linear(64, 1), std=1.0),
 76        ).to(self.device)
 77        self.actor_mean = nn.Sequential(
 78            layer_init(nn.Linear(obs_size, 64)),
 79            nn.Tanh(),
 80            layer_init(nn.Linear(64, 64)),
 81            nn.Tanh(),
 82            layer_init(nn.Linear(64, actions_size), std=0.01),
 83        ).to(self.device)
 84        self.actor_logstd = nn.Parameter(torch.zeros(1, actions_size)).to(self.device)
 85        if torch_compile:
 86            self.critic = torch.compile(self.critic)
 87            self.actor_mean = torch.compile(self.actor_mean)
 88
 89    def get_value(self, x: torch.Tensor) -> torch.Tensor:
 90        return self.critic(x)
 91
 92    def get_action(
 93        self, x: torch.Tensor, action: torch.Tensor | None = None
 94    ) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
 95        action_mean = self.actor_mean(x)
 96        action_logstd = self.actor_logstd.expand_as(action_mean)
 97        action_std = torch.exp(action_logstd)
 98        probs = Normal(action_mean, action_std)
 99        if action is None:
100            action = probs.sample()
101        return action, probs.log_prob(action).sum(1), probs.entropy().sum(1)
102
103    def get_action_and_value(
104        self, x: torch.Tensor, action: torch.Tensor | None = None
105    ) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor]:
106        action, log_probs, entropy = self.get_action(x, action=action)
107        value = self.get_value(x)
108        return action, log_probs, entropy, value

162class PPO:
163    """A Proximal Policy Optimization (PPO) agent that can handle both discrete and continuous action spaces.
164    Implementation coming from CleanRL: https://github.com/vwxyzjn/cleanrl/blob/master/cleanrl/ppo.py
165
166    It uses actor-critic networks under the hood and supports various features such as experience buffering,
167    advantage computation via GAE (Generalized Advantage Estimation), learning rate annealing, and optional
168    policy/value clipping in PPO updates.
169
170    Arguments:
171        observation_space (gymnasium.Space):
172            The space representing the type, shape, and bounds of observations.
173        action_space (gymnasium.Space):
174            The space representing the type, shape, and bounds of actions.
175        args (Args):
176            A container of hyperparameters controlling aspects like learning rate, gamma, gae_lambda, etc.
177        device (torch.device):
178            The Torch device (CPU or GPU) on which tensors should be allocated and computations performed.
179        name (str, optional):
180            An identifier for the agent.
181        torch_compile (bool, optional):
182            If True, attempts to compile relevant networks for optimization.
183
184    Attributes:
185        observation_space (gymnasium.Space):
186            Reference to the environment's observation space.
187        action_space (gymnasium.Space):
188            Reference to the environment’s action space.
189        discrete (bool):
190            Flag indicating if the action space is discrete or continuous.
191        actor_critic (Union[ContinuousActionAgent, DiscreteActionAgent]):
192            Neural network(s) used to produce actions and value estimates.
193        optimizer (torch.optim.Optimizer):
194            The optimizer for parameter updates (Adam in this case).
195        buffer_size (int):
196            Maximum capacity for the experience buffer.
197        states (torch.Tensor):
198            Buffer for storing observed states.
199        actions (torch.Tensor):
200            Buffer for storing performed actions.
201        rewards (torch.Tensor):
202            Buffer for storing immediate rewards received after actions.
203        log_probs (torch.Tensor):
204            Buffer for storing the log probabilities of actions.
205        dones (torch.Tensor):
206            Buffer for tracking whether an episode terminated at each step.
207        buffer_idx (int):
208            Current write index into the buffer.
209        valid_mask (torch.Tensor):
210            Mask to keep track of valid entries in the buffer (e.g. after the final state of an episode).
211        update_counter (int):
212            Number of updates performed so far.
213        num_updates (int):
214            Expected total number of training updates.
215
216    Methods:
217        reset_buffer():
218            Clears the buffer indices and resets validity masks.
219        store(state, action, reward, done):
220            Records a transition in the replay buffer, respecting capacity limits.
221        act(observation):
222            Returns a greedy (exploitation) action at inference time without exploration noise.
223        act_train(observation, global_step):
224            Sample an action from the policy using stochastic exploration during training.
225        compute_advantages(values, rewards):
226            Calculates advantages using GAE and resets advantages when episodes end.
227        update_step(global_step, writer):
228            Checks if the buffer is full and initiates a PPO update if needed.
229        _update_step(global_step, writer):
230            Performs the PPO training loop on the buffered experience, computing and applying gradients.
231        train(env, log_path, run_name):
232            Primary training loop for the agent, including data collection and periodic PPO updates.
233        seed(seed):
234            Sets the random seeds for reproducibility across all relevant libraries.
235        save_agent(save_path, name):
236            Saves the learned model parameters to a file for later reuse.
237        load_agent(load_path, name):
238            Loads model parameters from a file if it exists and updates the actor-critic accordingly.
239        update_lr(iteration):
240            Gradually reduces the learning rate over training to improve convergence stability."""

285        )

287        # Initialize buffer
288        self.buffer_size = self.args.batch_size
289
290        self.states = torch.zeros(
291            (self.buffer_size, self.observation_space.shape[0]), dtype=torch.float32

298                dtype=torch.float32,  # type: ignore
299            )

303            self.device

308
309        self.reset_buffer()
310        self.update_counter = 0
311        # How many updates will be done
312        self.num_updates = self.args.total_timesteps // self.args.batch_size

326    def store(
327        self,
328        state: np.ndarray | torch.Tensor,
329        action: np.ndarray | torch.Tensor | None = None,
330        reward: float | None = None,
331        done: bool | None = None,
332    ):
333        """Store transition in buffer"""
334        if self.buffer_idx >= self.buffer_size:
335            # Do this rather than raise an error cause the level_env might try to add twice before training due to the last obs
336            print(f"{self.name}: Buffer is full")
337            return
338
339        self.states[self.buffer_idx] = torch.FloatTensor(state)
340        if action is not None:
341            self.actions[self.buffer_idx] = torch.FloatTensor(action)
342            self.log_probs[self.buffer_idx] = self.last_logprob
343            self.valid_mask[self.buffer_idx] = True
344        else:
345            self.valid_mask[self.buffer_idx] = False
346
347        if reward is not None:
348            self.rewards[self.buffer_idx] = reward
349        else:
350            self.rewards[self.buffer_idx] = -1000
351        if done is not None:
352            self.dones[self.buffer_idx] = done
353
354        self.buffer_idx += 1

356    def act(self, observation: np.ndarray) -> int:
357        obs = torch.Tensor(np.expand_dims(observation, axis=0)).to(self.device)
358        with torch.no_grad():
359            # The act functions for the actor_critics sample from the distribution.
360            # But at inference we just want the max or the mean
361            if self.discrete:
362                action = (
363                    torch.argmax(self.actor_critic.actor(obs), dim=1).cpu().numpy()[0]
364                )
365            else:
366                action = self.actor_critic.actor_mean(obs).detach().cpu().numpy()
367        return action

369    def act_train(self, observation: np.ndarray, global_step: int) -> int:
370        """Act with exploration during training"""
371        obs = torch.Tensor(observation[np.newaxis, :]).to(self.device)
372        with torch.no_grad():
373            action, logprob, _ = self.actor_critic.get_action(obs)
374            self.last_logprob = logprob
375        return action.cpu().numpy()[0]

377    def compute_advantages(
378        self,
379        values: torch.Tensor,
380        rewards: torch.Tensor,
381    ) -> torch.Tensor:
382        """
383        Compute advantages using GAE
384        I don't use done masking, but just skip the points after done=True
385        """
386        advantages = torch.zeros(self.buffer_size).to(self.device)
387        gae = 0.0
388
389        for t in reversed(range(self.buffer_idx)):
390            # This is needed because of the end of episode storage
391            if self.valid_mask[t]:
392                if t < len(rewards) - 1 and not self.dones[t]:
393                    next_value = values[t + 1].item()
394                else:  # If done, we want to reset the gae as well (needed cause in the level env every stored sample is valid)
395                    next_value = 0.0
396                    gae = 0.0
397                delta = rewards[t] + self.args.gamma * next_value - values[t].item()  # type: ignore
398                gae = delta + self.args.gamma * self.args.gae_lambda * gae
399                advantages[t] = gae
400            else:
401                gae = 0.0  # Reset for a new episode
402
403        return advantages.unsqueeze(dim=-1)

412    def _update_step(self, global_step: int, writer: SummaryWriter | None):
413        """Perform PPO update"""
414        self.update_counter += 1
415        # Annealing the rate if instructed to do so.
416        if self.args.anneal_lr:
417            self.update_lr(iteration=self.update_counter)
418            if writer is not None:
419                writer.add_scalar(
420                    f"{self.name}/learning_rate",
421                    self.optimizer.param_groups[0]["lr"],
422                    global_step,
423                )
424
425        # Compute advantages
426        # ---------------------------
427        states = self.states.to(self.device)
428        with torch.no_grad():
429            old_values = self.actor_critic.get_value(states)
430        rewards = self.rewards.to(self.device)
431        # [batch_size, 1]
432        advantages = self.compute_advantages(old_values, rewards)
433
434        # Prepare batch by dropping elements that are after the end of an episode
435        # ---------------------------
436        valid_mask = self.valid_mask.to(self.device)
437
438        # NOTE we mask the advantages AFTER calculation cause before we need all the indexes
439        # But we have to do it before normalization
440        advantages = advantages[valid_mask]
441        if self.args.norm_adv:
442            advantages = (advantages - advantages.mean()) / (advantages.std() + 1e-8)
443
444        # [batch_size, 1]
445        actions = self.actions[self.valid_mask].to(self.device)
446        # [batch_size, state_dim]
447        states = states[valid_mask]
448        # [batch_size,]
449        old_log_probs = self.log_probs[valid_mask]
450        # [batch_size, 1]
451        old_values = old_values[valid_mask]
452
453        # [batch_size, 1]
454        returns = advantages + old_values
455        # ---------------------------
456
457        # PPO update for multiple epochs
458        # ================================
459        for update_ii in range(self.args.update_epochs):
460            # Shorter than batch size cause of end-of-episodes
461            data_length = int(self.valid_mask.sum())
462
463            # Generate random mini-batches
464            indices = np.random.permutation(data_length)
465
466            # Minibatches
467            # ================================
468            for start_idx in range(0, data_length, self.args.minibatch_size):
469                idx = indices[start_idx : start_idx + self.args.minibatch_size]
470
471                # Get mini-batch
472                mb_states = states[idx]
473                mb_actions = actions[idx]
474                mb_old_log_probs = old_log_probs[idx]
475                mb_advantages = advantages[idx]
476                mb_returns = returns[idx]
477                mb_values = old_values[idx]
478
479                # Get current action probabilities and values
480                (
481                    _,
482                    current_log_probs,
483                    entropy,
484                    current_values,
485                ) = self.actor_critic.get_action_and_value(mb_states, action=mb_actions)
486
487                # Compute ratio
488                # [minibatch_size,]
489                current_log_probs = torch.reshape(
490                    current_log_probs, mb_old_log_probs.shape
491                )
492                # [minibatch_size,]
493                logratio = current_log_probs - mb_old_log_probs
494                ratio = logratio.exp()
495                with torch.no_grad():
496                    # calculate approx_kl http://joschu.net/blog/kl-approx.html
497                    old_approx_kl = (-logratio).mean()
498                    approx_kl = ((ratio - 1) - logratio).mean()
499
500                # Policy loss
501                # [minibatch_size, 1]
502                ratio = torch.reshape(ratio, mb_advantages.shape)
503                pg_loss1 = -mb_advantages * ratio
504                pg_loss2 = -mb_advantages * torch.clamp(
505                    ratio, 1 - self.args.clip_coef, 1 + self.args.clip_coef
506                )
507                actor_loss = torch.max(pg_loss1, pg_loss2).mean()
508
509                # Critic loss
510                if self.args.clip_vloss:
511                    # [minibatch_size, 1]
512                    critic_loss_unclipped = (current_values - mb_returns) ** 2
513                    clipped_values = mb_values + torch.clamp(
514                        current_values - mb_values,
515                        -self.args.clip_coef,
516                        self.args.clip_coef,
517                    )
518                    critic_loss_clipped = (clipped_values - mb_returns) ** 2
519                    critic_loss_max = torch.max(
520                        critic_loss_unclipped, critic_loss_clipped
521                    )
522                    critic_loss = 0.5 * critic_loss_max.mean()
523                else:
524                    critic_loss = 0.5 * ((current_values - mb_returns) ** 2).mean()
525
526                # Total loss
527                entropy_loss = entropy.mean()
528                loss = (
529                    actor_loss
530                    + self.args.vf_coef * critic_loss
531                    - self.args.ent_coef * entropy_loss
532                )
533
534                # Update network
535                self.optimizer.zero_grad()
536                loss.backward()
537                nn.utils.clip_grad_norm_(
538                    self.actor_critic.parameters(), self.args.max_grad_norm
539                )
540                self.optimizer.step()
541
542                # Log metrics if writer is provided
543                if writer is not None:
544                    # There will be multiple values written for the same global step. Maybe I can do an avg over the minibatches?
545                    writer.add_scalar(
546                        f"{self.name}/value_loss", critic_loss.item(), global_step
547                    )
548                    writer.add_scalar(
549                        f"{self.name}/policy_loss", actor_loss.item(), global_step
550                    )
551                    writer.add_scalar(
552                        f"{self.name}/entropy", entropy_loss.item(), global_step
553                    )
554                    writer.add_scalar(
555                        f"{self.name}/old_approx_kl", old_approx_kl.item(), global_step
556                    )
557                    writer.add_scalar(
558                        f"{self.name}/approx_kl", approx_kl.item(), global_step
559                    )
560            # ================================
561        # ================================
562
563        # Log explained variance
564        y_pred, y_true = old_values.cpu().numpy(), returns.detach().cpu().numpy()
565        var_y = np.var(y_true)
566        explained_var = np.nan if var_y == 0 else 1 - np.var(y_true - y_pred) / var_y
567        if writer is not None:
568            writer.add_scalar(
569                f"{self.name}/explained_variance", explained_var, global_step
570            )
571
572        # Reset buffer after update
573        self.reset_buffer()

378    def compute_advantages(
379        self,
380        values: torch.Tensor,
381        rewards: torch.Tensor,
382    ) -> torch.Tensor:
383        """
384        Compute advantages using GAE
385        I don't use done masking, but just skip the points after done=True
386        """
387        advantages = torch.zeros(self.buffer_size).to(self.device)
388        gae = 0.0
389
390        for t in reversed(range(self.buffer_idx)):
391            # This is needed because of the end of episode storage
392            if self.valid_mask[t]:
393                if t < len(rewards) - 1 and not self.dones[t]:
394                    next_value = values[t + 1].item()
395                else:  # If done, we want to reset the gae as well (needed cause in the level env every stored sample is valid)
396                    next_value = 0.0
397                    gae = 0.0
398                delta = rewards[t] + self.args.gamma * next_value - values[t].item()  # type: ignore
399                gae = delta + self.args.gamma * self.args.gae_lambda * gae
400                advantages[t] = gae
401            else:
402                gae = 0.0  # Reset for a new episode
403
404        return advantages.unsqueeze(dim=-1)

413    def _update_step(self, global_step: int, writer: SummaryWriter | None):
414        """Perform PPO update"""
415        self.update_counter += 1
416        # Annealing the rate if instructed to do so.
417        if self.args.anneal_lr:
418            self.update_lr(iteration=self.update_counter)
419            if writer is not None:
420                writer.add_scalar(
421                    f"{self.name}/learning_rate",
422                    self.optimizer.param_groups[0]["lr"],
423                    global_step,
424                )
425
426        # Compute advantages
427        # ---------------------------
428        states = self.states.to(self.device)
429        with torch.no_grad():
430            old_values = self.actor_critic.get_value(states)
431        rewards = self.rewards.to(self.device)
432        # [batch_size, 1]
433        advantages = self.compute_advantages(old_values, rewards)
434
435        # Prepare batch by dropping elements that are after the end of an episode
436        # ---------------------------
437        valid_mask = self.valid_mask.to(self.device)
438
439        # NOTE we mask the advantages AFTER calculation cause before we need all the indexes
440        # But we have to do it before normalization
441        advantages = advantages[valid_mask]
442        if self.args.norm_adv:
443            advantages = (advantages - advantages.mean()) / (advantages.std() + 1e-8)
444
445        # [batch_size, 1]
446        actions = self.actions[self.valid_mask].to(self.device)
447        # [batch_size, state_dim]
448        states = states[valid_mask]
449        # [batch_size,]
450        old_log_probs = self.log_probs[valid_mask]
451        # [batch_size, 1]
452        old_values = old_values[valid_mask]
453
454        # [batch_size, 1]
455        returns = advantages + old_values
456        # ---------------------------
457
458        # PPO update for multiple epochs
459        # ================================
460        for update_ii in range(self.args.update_epochs):
461            # Shorter than batch size cause of end-of-episodes
462            data_length = int(self.valid_mask.sum())
463
464            # Generate random mini-batches
465            indices = np.random.permutation(data_length)
466
467            # Minibatches
468            # ================================
469            for start_idx in range(0, data_length, self.args.minibatch_size):
470                idx = indices[start_idx : start_idx + self.args.minibatch_size]
471
472                # Get mini-batch
473                mb_states = states[idx]
474                mb_actions = actions[idx]
475                mb_old_log_probs = old_log_probs[idx]
476                mb_advantages = advantages[idx]
477                mb_returns = returns[idx]
478                mb_values = old_values[idx]
479
480                # Get current action probabilities and values
481                (
482                    _,
483                    current_log_probs,
484                    entropy,
485                    current_values,
486                ) = self.actor_critic.get_action_and_value(mb_states, action=mb_actions)
487
488                # Compute ratio
489                # [minibatch_size,]
490                current_log_probs = torch.reshape(
491                    current_log_probs, mb_old_log_probs.shape
492                )
493                # [minibatch_size,]
494                logratio = current_log_probs - mb_old_log_probs
495                ratio = logratio.exp()
496                with torch.no_grad():
497                    # calculate approx_kl http://joschu.net/blog/kl-approx.html
498                    old_approx_kl = (-logratio).mean()
499                    approx_kl = ((ratio - 1) - logratio).mean()
500
501                # Policy loss
502                # [minibatch_size, 1]
503                ratio = torch.reshape(ratio, mb_advantages.shape)
504                pg_loss1 = -mb_advantages * ratio
505                pg_loss2 = -mb_advantages * torch.clamp(
506                    ratio, 1 - self.args.clip_coef, 1 + self.args.clip_coef
507                )
508                actor_loss = torch.max(pg_loss1, pg_loss2).mean()
509
510                # Critic loss
511                if self.args.clip_vloss:
512                    # [minibatch_size, 1]
513                    critic_loss_unclipped = (current_values - mb_returns) ** 2
514                    clipped_values = mb_values + torch.clamp(
515                        current_values - mb_values,
516                        -self.args.clip_coef,
517                        self.args.clip_coef,
518                    )
519                    critic_loss_clipped = (clipped_values - mb_returns) ** 2
520                    critic_loss_max = torch.max(
521                        critic_loss_unclipped, critic_loss_clipped
522                    )
523                    critic_loss = 0.5 * critic_loss_max.mean()
524                else:
525                    critic_loss = 0.5 * ((current_values - mb_returns) ** 2).mean()
526
527                # Total loss
528                entropy_loss = entropy.mean()
529                loss = (
530                    actor_loss
531                    + self.args.vf_coef * critic_loss
532                    - self.args.ent_coef * entropy_loss
533                )
534
535                # Update network
536                self.optimizer.zero_grad()
537                loss.backward()
538                nn.utils.clip_grad_norm_(
539                    self.actor_critic.parameters(), self.args.max_grad_norm
540                )
541                self.optimizer.step()
542
543                # Log metrics if writer is provided
544                if writer is not None:
545                    # There will be multiple values written for the same global step. Maybe I can do an avg over the minibatches?
546                    writer.add_scalar(
547                        f"{self.name}/value_loss", critic_loss.item(), global_step
548                    )
549                    writer.add_scalar(
550                        f"{self.name}/policy_loss", actor_loss.item(), global_step
551                    )
552                    writer.add_scalar(
553                        f"{self.name}/entropy", entropy_loss.item(), global_step
554                    )
555                    writer.add_scalar(
556                        f"{self.name}/old_approx_kl", old_approx_kl.item(), global_step
557                    )
558                    writer.add_scalar(
559                        f"{self.name}/approx_kl", approx_kl.item(), global_step
560                    )
561            # ================================
562        # ================================
563
564        # Log explained variance
565        y_pred, y_true = old_values.cpu().numpy(), returns.detach().cpu().numpy()
566        var_y = np.var(y_true)
567        explained_var = np.nan if var_y == 0 else 1 - np.var(y_true - y_pred) / var_y
568        if writer is not None:
569            writer.add_scalar(
570                f"{self.name}/explained_variance", explained_var, global_step
571            )
572
573        # Reset buffer after update
574        self.reset_buffer()

379        self,
380        values: torch.Tensor,
381        rewards: torch.Tensor,
382    ) -> torch.Tensor:
383        """
384        Compute advantages using GAE
385        I don't use done masking, but just skip the points after done=True
386        """
387        advantages = torch.zeros(self.buffer_size).to(self.device)
388        gae = 0.0
389
390        for t in reversed(range(self.buffer_idx)):
391            # This is needed because of the end of episode storage

396                    next_value = 0.0

399                gae = delta + self.args.gamma * self.args.gae_lambda * gae
400                advantages[t] = gae
401            else:
402                gae = 0.0  # Reset for a new episode
403
404        return advantages.unsqueeze(dim=-1)
