 58class Agent(BaseAgent):
 59    """A hierarchical agent class that implements a multi-level learning architecture.
 60
 61    This agent class creates and manages a hierarchical structure of learning agents organized in
 62    multiple levels (bottom, middle, top). Each level contains one or more agents that can
 63    communicate and make decisions. The hierarchy allows for both independent and coordinated
 64    learning across different levels.
 65
 66        env (ParallelEnv): The parallel environment that the agent will interact with.
 67        args (None | Args, optional): Configuration arguments for the agent. If None, default
 68            Args will be used. Defaults to None.
 69
 70    Attributes:
 71        args (Args): Configuration arguments for the agent.
 72        device (torch.device): The device (CPU/GPU) where computations will be performed.
 73        env (ParallelEnv): The environment the agent interacts with.
 74        hierarchy (Hierarchy): The hierarchical structure containing all agent levels.
 75        num_env_agents (int): Number of agents in the environment.
 76        bottom_env_links (dict): Mapping between bottom level agents and environment agents.
 77        mid_bottom_links (dict): Mapping between middle level and bottom level agents.
 78        top_mid_links (dict): Mapping between top level and middle level agents.
 79
 80    Methods:
 81        seed(seed): Sets random seeds for reproducibility.
 82        train(env, log_path, run_name): Trains the hierarchical agent.
 83        save_agent(save_path, name): Saves the agent's hierarchy to disk.
 84        load_agent(load_path, name): Loads a trained agent from disk.
 85        act(observation): Returns action based on given observation.
 86
 87    Note:
 88        The hierarchy consists of three levels:
 89        - Bottom level: Contains individual PPO agents for each environment agent
 90        - Middle level: Contains MAPPO agents that coordinate groups of bottom agents
 91        - Top level: Contains a single MAPPO agent that coordinates middle level agents
 92
 93    Example:
 94        >>> env = ParallelEnv()
 95        >>> args = Args()
 96        >>> agent = Agent(env, args)
 97        >>> agent.train()
 98
 99        This is the structure that the agent would have for an environment with 4 agents.
100        Instantiated hierarchy:
101        └── top_mappo
102            ├── middle_mappo_1
103            │   ├── agent_0
104            │   │   └── agent_0 (env)
105            │   └── agent_1
106            │       └── agent_1 (env)
107            └── middle_mappo_2
108                ├── agent_2
109                │   └── agent_2 (env)
110                └── agent_3
111                    └── agent_3 (env)
112    """
113
114    def __init__(self, env: ParallelEnv, args: None | Args = None) -> None:
115        if args is None:
116            self.args: Args = Args()
117        else:
118            self.args: Args = args
119
120        if self.args.cuda >= 0:
121            self.device = torch.device(
122                f"cuda:{self.args.cuda}" if torch.cuda.is_available() else "cpu"
123            )
124        self.env = env
125        self.seed(self.args.seed)
126
127        # Create hierarchy
128        self.hierarchy = Hierarchy()
129        self.num_env_agents = len(env.possible_agents)
130
131        # ---------------------------
132        # Define agent names
133        name_top_agent = "top_mappo"
134        bottom_agent_names = [f"agent_{i}" for i in range(self.num_env_agents)]
135        middle_names = ["middle_mappo_1", "middle_mappo_2"]
136
137        # Define hierarchy links
138        self.bottom_env_links = {
139            agent_name: [agent_name] for agent_name in bottom_agent_names
140        }
141        self.mid_bottom_links = {
142            middle_names[0]: bottom_agent_names[: int(self.num_env_agents / 2)],
143            middle_names[1]: bottom_agent_names[int(self.num_env_agents / 2) :],
144        }
145        self.top_mid_links = {name_top_agent: middle_names}
146        # ---------------------------
147
148        # Configure bottom level
149        # ================================
150        # Configure agents
151        bottom_agents = [
152            AgentConfig(
153                name=name,
154                communication_space=GymDict(
155                    {name: Box(-np.inf, np.inf, shape=[self.args.comm_size])}  # type: ignore
156                )
157                if (hasattr(self.args, "learn_comm") and self.args.learn_comm)  # type: ignore
158                else GymDict({name: env.observation_space(name)}),
159                agent_class=PPO,
160                agent_kwargs={"args": self.args.ppo_args},
161                device=self.device,
162            )
163            for name in bottom_agent_names
164        ]
165
166        # Configure bottom level action space
167        bottom_level_act_shape = 2
168        bottom_level_action_space = {}
169        for hl_agent, agent_names in self.mid_bottom_links.items():
170            # Continuous action space!
171            agent_spaces = {
172                name: Box(-np.inf, np.inf, shape=[bottom_level_act_shape])
173                for name in agent_names
174            }
175            bottom_level_action_space[hl_agent] = GymDict(agent_spaces)  # type: ignore
176        bottom_level_action_space = GymDict(bottom_level_action_space)
177
178        # Configure level
179        bottom_level = LevelConfig(
180            name="bottom",
181            agents=bottom_agents,
182            uplinks=self.mid_bottom_links,
183            downlinks=self.bottom_env_links,
184            action_frequency=self.args.ifreq_bottom,
185            trace_type="full" if self.args.save_all_trace else "reward",
186            concat_obs=True,
187            action_space=bottom_level_action_space,
188            env=env,
189        )
190        self.hierarchy.add_level_config(bottom_level)
191        # ================================
192
193        # Configure middle level
194        # ================================
195        # Configure agents
196        middle_agents = []
197        prev_level = self.hierarchy.levels[-1]
198        for agent_name in middle_names:
199            # Make agent comm space
200            # ---------------------------
201            if hasattr(self.args, "learn_comm") and self.args.learn_comm:  # type: ignore
202                agent_comm_space = GymDict(
203                    {agent_name: Box(-np.inf, np.inf, shape=[self.args.comm_size])}  # type: ignore
204                )
205            else:
206                # In this case, we just concatenate the observations from the level l-1
207                prev_lev_obs_spaces = list(
208                    prev_level.observation_space(agent_name).values()  # type: ignore
209                )
210                agent_comm_space = GymDict(
211                    {agent_name: merge_spaces(prev_lev_obs_spaces)}
212                )
213            # ---------------------------
214
215            # Agent config
216            middle_agents.append(
217                AgentConfig(
218                    name=agent_name,
219                    communication_space=agent_comm_space,
220                    agent_class=MAPPO,
221                    agent_kwargs={"args": self.args.mappo_args},
222                    device=self.device,
223                )
224            )
225
226        # Configure middle level action space
227        middle_level_act_shape = 2
228        middle_level_action_space = {}
229        for hl_agent, agent_names in self.top_mid_links.items():
230            # Continuous action space!
231            agent_spaces = {
232                name: Box(-np.inf, np.inf, shape=[middle_level_act_shape])
233                for name in agent_names
234            }
235            middle_level_action_space[hl_agent] = GymDict(agent_spaces)  # type: ignore
236        middle_level_action_space = GymDict(middle_level_action_space)
237
238        # Configure level
239        middle_level = LevelConfig(
240            name="middle",
241            agents=middle_agents,
242            uplinks=self.top_mid_links,
243            downlinks=self.mid_bottom_links,
244            action_frequency=self.args.ifreq_mid,
245            trace_type="full" if self.args.save_all_trace else "reward",
246            concat_obs=True,
247            action_space=middle_level_action_space,
248            env=self.hierarchy.levels[-1],
249        )
250        self.hierarchy.add_level_config(middle_level)
251        # ================================
252
253        # Configure top level
254        # ================================
255        top_agent = AgentConfig(
256            name=name_top_agent,
257            communication_space=None,
258            agent_class=MAPPO,
259            agent_kwargs={"args": self.args.mappo_args},
260            device=self.device,
261        )
262
263        top_level = LevelConfig(
264            name="top",
265            agents=[top_agent],
266            uplinks=None,
267            downlinks=self.top_mid_links,
268            action_frequency=self.args.ifreq_top,
269            trace_type="full" if self.args.save_all_trace else "reward",
270            env=self.hierarchy.levels[-1],
271        )
272        self.hierarchy.add_level_config(top_level)
273        # ================================
274
275        print("Instantiated hierarchy:")
276        self.hierarchy.print_tree()

297    def train(
298        self,
299        env: ParallelEnv | None = None,
300        log_path: Path | str | None = None,
301        run_name: str | None = None,
302    ):
303        """
304        Train the hierarchical agent using the hierarchy.
305
306        This method implements the training loop for the hierarchical agent. It sets up logging,
307        initializes the environment, and runs the training episodes until the specified total
308        timesteps are reached. Creates a directory structure for logging under log_path/run_name,
309        saves hyperparameters and interface level information as JSON files, uses tensorboard
310        for logging training metrics. Training continues until total_timesteps (specified in
311        self.args) is reached.
312
313        Args:
314            env (ParallelEnv | None, optional): The environment to train on. If None, uses
315                the environment stored in self.env passed at init. Defaults to None.
316            log_path (Path | str | None, optional): Path where to save the training logs.
317                If None, saves to "runs" directory. Defaults to None.
318            run_name (str | None, optional): Name for the training run. If None, generates
319                a name using experiment name, seed and timestamp. Defaults to None.
320
321        Returns:
322            None
323
324        Note:
325            - Creates a directory structure for logging under log_path/run_name
326            - Saves hyperparameters and interface level information as JSON files
327            - Uses tensorboard for logging training metrics
328            - Training continues until total_timesteps (specified in self.args) is reached
329        """
330        if env is None:
331            env = self.env
332        self.hierarchy.connect(env)
333
334        # Prepare for logging
335        if run_name is None:
336            run_name = f"{self.args.exp_name}__{self.args.seed}__{int(time.time())}"
337
338        if log_path is None:
339            log_path = Path("runs")
340        else:
341            log_path = Path(log_path)
342
343        if not (log_path / run_name).exists():
344            os.makedirs(log_path / run_name)
345
346        args_dict = self.args.to_dict()  # type: ignore
347        with open(log_path / run_name / "params.json", "w") as f:
348            json.dump(args_dict, f, indent=4)
349
350        interface_level_name = {"name": self.hierarchy.interface_level.name}
351        with open(log_path / run_name / "interface_level.json", "w") as f:
352            json.dump(interface_level_name, f, indent=4)
353
354        save_path = log_path / run_name / "training"
355
356        writer = SummaryWriter(save_path / "tboard")
357        writer.add_text(
358            "hyperparameters",
359            "|param|value|\n|-|-|\n%s"
360            % (
361                "\n".join(
362                    [f"|{key}|{value}|" for key, value in vars(self.args).items()]
363                )
364            ),
365        )
366
367        self.hierarchy.set_logger(logger=writer, save_path=save_path)
368
369        # Training loop
370        # ---------------------------
371        done = False
372        episode = 0
373        self.hierarchy.reset()
374
375        for global_step in tqdm(
376            range(self.args.total_timesteps), desc="Training step:"
377        ):
378            if done:
379                done = False
380                self.hierarchy.reset()
381                episode += 1
382
383            # Training and environment interaction is handled by the hierarchy
384            _, reward, terminated, truncated, _ = self.hierarchy.step(action=None)
385
386            if any(terminated.values()) or any(truncated.values()):
387                done = True
388        # ---------------------------
389
390        # Final cleanup
391        self.hierarchy.reset()
392        writer.close()

412
413        This method loads a previously trained agent's model from the specified path.

This agent class creates and manages a hierarchical structure of learning agents organized in multiple levels (bottom, middle, top). Each level contains one or more agents that can communicate and make decisions. The hierarchy allows for both independent and coordinated learning across different levels.

    Args will be used. Defaults to None.

Methods: seed(seed): Sets random seeds for reproducibility. train(env, log_path, run_name): Trains the hierarchical agent. save_agent(save_path, name): Saves the agent's hierarchy to disk. load_agent(load_path, name): Loads a trained agent from disk. act(observation): Returns action based on given observation.
