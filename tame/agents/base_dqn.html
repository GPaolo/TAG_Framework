 67def linear_schedule(start_e: float, end_e: float, duration: int, t: int):
 68    slope = (end_e - start_e) / duration
 69    return max(slope * t + start_e, end_e)

 72class DQN:
 73    """Deep Q-Network (DQN) implementation for reinforcement learning.
 74    Implementation coming from CleanRL: https://github.com/vwxyzjn/cleanrl/blob/master/cleanrl/dqn.py
 75
 76    This class implements a DQN agent that can learn to make decisions in environments
 77    with discrete action spaces and continuous observation spaces. It includes features
 78    such as experience replay, target network, and epsilon-greedy exploration.
 79
 80    Args:
 81        observation_space (gymnasium.Space): The observation space of the environment.
 82        action_space (gymnasium.Space): The action space of the environment.
 83        args (Args): Configuration parameters for the DQN agent.
 84        device (torch.device): The device (CPU/GPU) to run the computations on.
 85        name (str, optional): Name identifier for the DQN agent. Defaults to "dqn".
 86        torch_compile (bool, optional): Whether to compile the networks using torch.compile.
 87            Defaults to True.
 88
 89    Attributes:
 90        observation_space (gymnasium.Space): The observation space of the environment.
 91        action_space (gymnasium.Space): The action space of the environment.
 92        obs_size (int): The size of the observation space.
 93        actions_n (int): The number of possible actions.
 94        q_net (QNetwork): The main Q-network for action value estimation.
 95        target_net (QNetwork): The target network for stable learning.
 96        rb (ReplayBuffer): Buffer for storing and sampling experiences.
 97        optimizer (torch.optim.Adam): The optimizer for training the Q-network.
 98
 99    Methods:
100        seed(seed): Sets random seeds for reproducibility.
101        update_qnet(): Performs one gradient step on the Q-network.
102        update_target_net(): Updates the target network parameters.
103        update_step(global_step, writer): Performs a training step.
104        train(env, log_path, run_name): Trains the DQN agent in the given environment.
105        act_train(observation, global_step): Selects actions during training (with exploration).
106        act(observation): Selects actions using the trained policy (without exploration).
107        store(state, action, reward, done): Stores transitions in the replay buffer.
108        save_agent(save_path, name): Saves the trained model to disk.
109        load_agent(load_path, name): Loads a trained model from disk.
110
111    Note:
112        The implementation assumes a discrete action space and a continuous observation
113        space. The environment must be compatible with the Gymnasium interface.
114    """
115
116    def __init__(
117        self,
118        observation_space: gymnasium.Space,
119        action_space: gymnasium.Space,
120        args: Args,
121        device: torch.device,
122        name: str = "dqn",
123        torch_compile: bool = True,
124    ):
125        self.observation_space = observation_space
126        self.action_space = action_space
127        assert isinstance(self.observation_space, gymnasium.spaces.Box)
128        assert isinstance(self.action_space, gymnasium.spaces.Discrete)
129
130        self.obs_size = self.observation_space.shape[0]
131        self.actions_n = int(self.action_space.n)
132        self.args = args
133        self.name = name
134        self.device = device
135        self.seed(seed=self.args.seed)
136
137        self.q_net = QNetwork(
138            obs_size=self.obs_size,
139            actions_n=self.actions_n,
140            device=self.device,
141            torch_compile=torch_compile,
142        )
143
144        self.target_net = QNetwork(
145            obs_size=self.obs_size,
146            actions_n=self.actions_n,
147            device=self.device,
148            torch_compile=torch_compile,
149        )
150
151        self.rb = ReplayBuffer(
152            capacity=self.args.buffer_size, seed=self.args.seed, device=self.device
153        )
154        self.rb_idx = 0
155        self.optimizer = optim.Adam(self.q_net.parameters(), lr=self.args.learning_rate)  # type: ignore
156        self.target_net.load_state_dict(self.q_net.state_dict())
157
158    def seed(self, seed):
159        random.seed(seed)
160        np.random.seed(seed)
161        torch.manual_seed(seed)
162        torch.backends.cudnn.deterministic = self.args.torch_deterministic
163
164    def update_qnet(self) -> Tuple[np.ndarray, np.ndarray]:
165        """This function performs one gradient step on the Qnet"""
166        # Get Loss
167        # ---------------------------
168        data, _ = self.rb.sample(self.args.batch_size)
169        with torch.no_grad():
170            target_max, _ = self.target_net(data.next_observations).max(dim=-1)
171            td_target = data.rewards.flatten() + self.args.gamma * target_max * (
172                1 - data.dones.flatten()
173            )
174        old_val = (
175            self.q_net(data.observations)
176            .gather(-1, data.actions.to(torch.int64))
177            .squeeze()
178        )
179        loss = F.mse_loss(target=td_target, input=old_val)
180        # ---------------------------
181
182        # optimize the model
183        # ---------------------------
184        self.optimizer.zero_grad()
185        loss.backward()
186        self.optimizer.step()
187        # -------------------------
188        return loss.detach().cpu().numpy(), old_val.detach().cpu().numpy()
189
190    def update_target_net(self):
191        """This function updates the target network"""
192        for target_net_p, q_net_p in zip(
193            self.target_net.parameters(), self.q_net.parameters()
194        ):
195            target_net_p.data.copy_(
196                self.args.tau * q_net_p.data + (1.0 - self.args.tau) * target_net_p.data
197            )
198
199    def update_step(self, global_step: int, writer: None | SummaryWriter):
200        """This function performs a train step"""
201        if global_step > self.args.learning_starts:
202            if global_step % self.args.train_frequency == 0:
203                loss, q_val = self.update_qnet()
204
205                # Log
206                # ---------------------------
207                if global_step % 100 == 0 and writer is not None:
208                    writer.add_scalar(f"{self.name}/td_loss", loss, global_step)
209                    writer.add_scalar(
210                        f"{self.name}/q_values", np.mean(q_val), global_step
211                    )
212                # --------------------------
213
214            if global_step % self.args.target_network_frequency == 0:
215                self.update_target_net()
216
217    def train(
218        self, env: Any, log_path: Path | str | None = None, run_name: str | None = None
219    ):
220        if run_name is None:
221            run_name = f"{self.args.exp_name}__{self.args.seed}__{int(time.time())}"
222
223        if log_path is None:
224            log_path = Path("runs")
225        else:
226            log_path = Path(log_path)
227
228        writer = SummaryWriter(log_path / run_name)
229        writer.add_text(
230            "hyperparameters",
231            "|param|value|\n|-|-|\n%s"
232            % (
233                "\n".join(
234                    [f"|{key}|{value}|" for key, value in vars(self.args).items()]
235                )
236            ),
237        )
238
239        # Start the game
240        obs, _ = env.reset(seed=self.args.seed)
241        done = False
242        episodic_return = 0
243        for global_step in tqdm(
244            range(self.args.total_timesteps), desc="Training step:"
245        ):
246            # Reset
247            # ---------------------------
248            if done:
249                obs, infos = env.reset(seed=self.args.seed)
250                done = False
251                episodic_return = 0
252            # ---------------------------
253
254            # Get actions
255            # ---------------------------
256            action = self.act_train(observation=obs, global_step=global_step)
257            # ---------------------------
258
259            # Perform action
260            # ---------------------------
261            next_obs, reward, terminated, truncated, info = env.step(action)
262            # ---------------------------
263
264            # Save data and log
265            # ---------------------------
266            episodic_return += reward
267            done = terminated or truncated
268            self.store(state=obs, action=action, reward=reward, done=done)
269
270            if done:
271                self.store(state=next_obs)  # store next obs
272                print(f"global_step={global_step}, - Ep. return={episodic_return}")
273                writer.add_scalar(f"returns/{self.name}", episodic_return, global_step)
274            # ---------------------------
275
276            obs = next_obs
277
278            self.update_step(global_step=global_step, writer=writer)
279
280        if self.args.save_model:
281            save_path = log_path / run_name / "trained_policy.pth"
282            torch.save(self.q_net.state_dict(), save_path)
283            print(f"model saved to {save_path}")
284
285        writer.close()
286        self.trained = True

288    def act_train(
289        self,
290        observation: np.ndarray,
291        global_step: int,
292    ) -> int:
293        # Perform random exploration
294        epsilon = linear_schedule(
295            self.args.start_e,
296            self.args.end_e,
297            int(self.args.exploration_fraction * self.args.total_timesteps),
298            global_step,
299        )
300
301        if random.random() < epsilon:
302            action = self.action_space.sample()
303        else:
304            q_values = self.q_net(torch.Tensor([observation]).to(self.device))
305            action = torch.argmax(q_values, dim=1).cpu().numpy()[0]
306        return action
307
308    def act(self, observation: np.ndarray) -> int:
309        q_values = self.q_net(torch.Tensor(np.array([observation])).to(self.device))
310        action = torch.argmax(q_values, dim=1).cpu().numpy()[0]
311        return action

313    def store(self, state, action=None, reward=None, done=None):
314        self.rb.push(
315            state=state, idx=self.rb_idx, action=action, reward=reward, done=done
316        )
317        self.rb_idx += 1

320        save_path = Path(save_path)
321        if name is None:
322            name = "trained_model"
323        model_save_path = save_path / "models" / f"{name}.pth"

327        if self.load_agent(save_path, name=f"{name}.pth"):
328            print(f"model saved to {model_save_path}")
329        else:
330            print("Could not save the model!")

Base class for all neural network modules.

Your models should also subclass this class.

Modules can also contain other Modules, allowing them to be nested in a tree structure. You can assign the submodules as regular attributes::

import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    def __init__(self) -> None:
        super().__init__()
        self.conv1 = nn.Conv2d(1, 20, 5)
        self.conv2 = nn.Conv2d(20, 20, 5)

    def forward(self, x):
        x = F.relu(self.conv1(x))
        return F.relu(self.conv2(x))

Submodules assigned in this way will be registered, and will also have their parameters converted when you call to(), etc.

As per the example above, an __init__() call to the parent class must be made before assignment on the child.

:ivar training: Boolean represents whether this module is in training or evaluation mode. :vartype training: bool

165    def update_qnet(self) -> Tuple[np.ndarray, np.ndarray]:
166        """This function performs one gradient step on the Qnet"""
167        # Get Loss
168        # ---------------------------
169        data, _ = self.rb.sample(self.args.batch_size)
170        with torch.no_grad():
171            target_max, _ = self.target_net(data.next_observations).max(dim=-1)
172            td_target = data.rewards.flatten() + self.args.gamma * target_max * (
173                1 - data.dones.flatten()
174            )
175        old_val = (
176            self.q_net(data.observations)
177            .gather(-1, data.actions.to(torch.int64))
178            .squeeze()
179        )
180        loss = F.mse_loss(target=td_target, input=old_val)
181        # ---------------------------
182
183        # optimize the model
184        # ---------------------------
185        self.optimizer.zero_grad()
186        loss.backward()
187        self.optimizer.step()
188        # -------------------------
189        return loss.detach().cpu().numpy(), old_val.detach().cpu().numpy()

197                self.args.tau * q_net_p.data + (1.0 - self.args.tau) * target_net_p.data

215            if global_step % self.args.target_network_frequency == 0:

Deep Q-Network (DQN) implementation for reinforcement learning. Implementation coming from CleanRL: https://github.com/vwxyzjn/cleanrl/blob/master/cleanrl/dqn.py

This class implements a DQN agent that can learn to make decisions in environments with discrete action spaces and continuous observation spaces. It includes features such as experience replay, target network, and epsilon-greedy exploration.

Args: observation_space (gymnasium.Space): The observation space of the environment. action_space (gymnasium.Space): The action space of the environment. args (Args): Configuration parameters for the DQN agent. device (torch.device): The device (CPU/GPU) to run the computations on. name (str, optional): Name identifier for the DQN agent. Defaults to "dqn". torch_compile (bool, optional): Whether to compile the networks using torch.compile. Defaults to True.

Attributes: observation_space (gymnasium.Space): The observation space of the environment. action_space (gymnasium.Space): The action space of the environment. obs_size (int): The size of the observation space. actions_n (int): The number of possible actions. q_net (QNetwork): The main Q-network for action value estimation. target_net (QNetwork): The target network for stable learning. rb (ReplayBuffer): Buffer for storing and sampling experiences. optimizer (torch.optim.Adam): The optimizer for training the Q-network.

Methods: seed(seed): Sets random seeds for reproducibility. update_qnet(): Performs one gradient step on the Q-network. update_target_net(): Updates the target network parameters. update_step(global_step, writer): Performs a training step. train(env, log_path, run_name): Trains the DQN agent in the given environment. act_train(observation, global_step): Selects actions during training (with exploration). act(observation): Selects actions using the trained policy (without exploration). store(state, action, reward, done): Stores transitions in the replay buffer. save_agent(save_path, name): Saves the trained model to disk. load_agent(load_path, name): Loads a trained model from disk.

Note: The implementation assumes a discrete action space and a continuous observation space. The environment must be compatible with the Gymnasium interface.
