 19def is_nested_dict(
 20    obj: Dict[str, np.ndarray] | Dict[str, Dict[str, np.ndarray]],
 21) -> TypeGuard[Dict[str, Dict[str, np.ndarray]]]:
 22    """Check if a dictionary is a nested dictionary.
 23
 24    This function determines if a dictionary contains dictionaries as values.
 25    It checks whether all values in the input dictionary are themselves dictionaries.
 26
 27    Args:
 28        obj: A dictionary that contains either numpy arrays or dictionaries of numpy arrays.
 29            Can be either Dict[str, np.ndarray] or Dict[str, Dict[str, np.ndarray]].
 30
 31    Returns:
 32        TypeGuard[Dict[str, Dict[str, np.ndarray]]]: True if all values in the dictionary
 33            are dictionaries, False otherwise. The TypeGuard ensures type narrowing
 34            in a type checker.
 35
 36    Examples:
 37        >>> d1 = {"a": {"x": np.array([1])}}  # Nested dict
 38        >>> d2 = {"a": np.array([1])}         # Flat dict
 39        >>> is_nested_dict(d1)
 40        True
 41        >>> is_nested_dict(d2)
 42        False
 43    """
 44    return all(isinstance(v, dict) for v in obj.values())

 47class LevelEnv:
 48    """A hierarchical environment wrapper that manages multiple agents at a specific level.
 49
 50    This class creates a hierarchical layer of agents that can interact with both higher and lower levels
 51    in a hierarchical reinforcement learning setup. It handles the communication between levels,
 52    action execution, and agent management.
 53
 54    Attributes:
 55        verbose (int): Verbosity level for logging.
 56        concat_obs (bool): Whether to concatenate top observations with lower messages.
 57        obs_from_bottom (dict): Storage for observations from lower level.
 58        agents (Dict[str, LevelAgent]): Dictionary of agents at this level.
 59        env (Any): The lower level environment.
 60        uplinks (Dict[str, list]): Mapping of higher level agents to their controlled agents.
 61        downlinks (Dict[str, list]): Mapping of agents to their controlled lower level agents.
 62        action_ifreq (int): Action frequency relative to lower level steps.
 63        trace (object): Trace object for logging episode data.
 64        level_ts (int): Total timesteps at this level.
 65        episode_ts (int): Timesteps in current episode.
 66        episode_idx (int): Current episode index.
 67        name (str): Identifier for this level.
 68        logger (SummaryWriter): Logger for metrics.
 69        save_path (Path): Path for saving model data.
 70        observation_spaces (GymDict): Observation spaces for each agent.
 71        action_spaces (GymDict): Action spaces for each agent.
 72        episode_returns (dict): Cumulative returns for current episode.
 73        env_actions (dict): Latest actions sent to lower level.
 74
 75    Methods:
 76        action_space(agent_name): Returns the action space for a specific agent.
 77        observation_space(agent_name): Returns the observation space for a specific agent.
 78        seed(seed): Sets random seeds for reproducibility.
 79        set_logger(logger, save_path): Configures logging and saving directories.
 80        reset(): Resets the level and returns initial observations.
 81        step(action, training): Executes one step of the environment.
 82        update_step(): Performs learning updates for agents.
 83        save(save_path): Saves all agents' models.
 84        load(load_path): Loads all agents' models.
 85        make_message(...): Creates messages for higher level from lower level data.
 86        act(observation, training): Generates actions for lower level from agents.
 87    """

 89    def __init__(
 90        self,
 91        agents: Dict[str, LevelAgent],
 92        uplinks: Dict[str, list] | None,
 93        downlinks: Dict[str, list],
 94        env: Any,
 95        action_space: GymDict | None = None,  # {str: Box | Discrete}
 96        concat_obs: bool = False,
 97        trace_type: str = "none",
 98        name: str = "level_0",
 99        verbose: int = 0,
100        action_ifreq: int = 1,
101    ):
102        """This class wraps a level of agents to behave like a gym environment.
103
104        Manages hierarchical interactions between agents at different levels by handling observations,
105        actions and message passing between levels.
106
107        Args:
108            agents (Dict[str, LevelAgent]): Dictionary of agents at this level, keyed by agent name
109            uplinks (Dict[str, list] | None): Mapping of how each agent links to higher level agents. Format: {higher_agent: [lower_agents]}
110            downlinks (Dict[str, list]): Mapping of how each agent links to lower level agents. Format: {agent: [lower_agents]}
111            env (Any): The environment this level interacts with (typically lower level env)
112            action_space (GymDict | None, optional): Custom action space. Only valid if concat_obs=True. Defaults to None.
113            concat_obs (bool, optional): If True, concatenates top observations with lower messages [msg, obs].
114                                       If False, agent only sees top observations. Defaults to False.
115            trace_type (str, optional): Type of execution trace to maintain. Must be one of trace_types. Defaults to "none".
116            name (str, optional): Name identifier for this level. Defaults to "level_0".
117            verbose (int, optional): Verbosity level for logging. Defaults to 0.
118            action_ifreq (int, optional): Action interval frequency relative to lower level steps.
119                                         Minimum value is 1. Defaults to 1.
120
121            ValueError: If trace_type is invalid or if action_space is provided without concat_obs=True
122
123        Attributes:
124            verbose (int): Verbosity level
125            concat_obs (bool): Whether to concatenate observations
126            obs_from_bottom (dict): Observations received from lower level
127            agents (dict): The level's agents
128            env (Any): Lower level environment
129            uplinks (dict): Upper level connections
130            downlinks (dict): Lower level connections
131            action_ifreq (int): Action interval frequency
132            trace: Trace object for logging execution
133            level_ts (int): Global timestep counter (not reset between episodes)
134            episode_ts (int): Episode timestep counter (reset each episode)
135            episode_idx (int): Current episode index
136            name (str): Level name identifier
137            logger: Logger object
138            save_path: Path for saving data
139            observation_spaces: Observation spaces for the level
140            action_spaces: Action spaces for the level"""
141        self.verbose = verbose
142        self.concat_obs = concat_obs
143        self.obs_from_bottom = {}
144        self.agents = agents
145        self.env = env
146        self.uplinks = uplinks
147        self.downlinks = downlinks
148        self.action_ifreq = max(1, action_ifreq)
149        if trace_type not in trace_types:
150            raise ValueError(
151                f"Wrong trace type specified for {name}. Given: {trace_type} - Available: {trace_types.keys()}"
152            )
153        self.trace = trace_types[trace_type]
154        if self.trace is not None:
155            self.trace = self.trace()
156        self.level_ts = 0  # This one is not reset
157        self.episode_ts = 0  # This one is reset
158        self.episode_idx = -1
159        self.name = name

161        self.logger = None
162        self.save_path = None
163        self.observation_spaces = self._make_observation_space()
164
165        if action_space is not None:
166            if not self.concat_obs:
167                raise ValueError(
168                    f"{name} - You can only specify an action space if you pass concat_obs=True"
169                )
170            self.action_spaces = action_space
171        else:
172            self.action_spaces = self._make_action_space()
173        print(f"Created level {self.name} with agents: {list(self.downlinks.keys())}")

205    def _make_observation_space(self) -> GymDict:  # {str: Box | Discrete}
206        """Make observation spaces based on the communication spaces of the agents.
207
208        This method creates a dictionary of observation spaces for higher-level agents based on
209        the communication spaces of their connected lower-level agents defined in uplinks.
210
211        Returns:
212            GymDict: A dictionary containing observation spaces for each higher-level agent.
213                    The structure is {hl_agent: {agent_name: communication_space}}.
214                    If no uplinks are defined, returns an empty GymDict.
215
216        Raises:
217            ValueError: If any agent in uplinks doesn't have a defined communication_space attribute.
218
219        Notes:
220            - Requires self.uplinks to be defined as a dictionary mapping higher-level agents
221              to lists of lower-level agent names.
222            - Each agent must have a communication_space attribute defining its observation space.
223            - If self.verbose >= 2, prints the observation space structure.
224        """
225        observation_space = {}
226        if self.uplinks is None:
227            return GymDict(observation_space)
228        for hl_agent, agent_names in self.uplinks.items():
229            agent_spaces = {}
230            for agent_name in agent_names:
231                if not hasattr(self.agents[agent_name], "communication_space"):
232                    raise ValueError(
233                        f"Agent {agent_name} does not have an communication_space"
234                    )
235                agent_spaces.update(self.agents[agent_name].communication_space)  # type: ignore
236            observation_space[hl_agent] = GymDict(agent_spaces)
237
238        if self.verbose >= 2:
239            print(f"{self.name} -  observation space")
240            print(f"\t {observation_space}")
241        return GymDict(observation_space)

243    def _make_action_space(self) -> GymDict:  # {str: Box | Discrete}
244        """Make action space based on the observation spaces of the agents.
245
246        This method creates an action space dictionary where each high-level agent's action space
247        is defined by the observation spaces of its associated lower-level agents.
248
249        Returns:
250            GymDict: A dictionary mapping high-level agent names to their action spaces.
251                Each action space is a GymDict containing the observation spaces of the
252                controlled lower-level agents. If no uplinks are defined, returns an empty GymDict.
253
254        Raises:
255            ValueError: If any of the controlled agents does not have an observation_space attribute.
256
257        Note:
258            The action space of each high-level agent is constructed by combining the
259            observation spaces of all lower-level agents it controls, as defined in self.uplinks.
260        """
261        action_space = {}
262        if self.uplinks is None:
263            return GymDict(action_space)
264        for hl_agent, agent_names in self.uplinks.items():
265            agent_spaces = {}
266            for agent_name in agent_names:
267                if not hasattr(self.agents[agent_name], "observation_space"):
268                    raise ValueError(
269                        f"Agent {agent_name} does not have an observation_space"
270                    )
271                agent_spaces.update(self.agents[agent_name].observation_space)
272            action_space[hl_agent] = GymDict(agent_spaces)
273        if self.verbose >= 2:
274            print(f"{self.name} -  action space")
275            print(f"\t {action_space}")
276        return GymDict(action_space)

330    def reset(self) -> Tuple[Dict[str, Dict[str, np.ndarray]] | None, dict | None]:
331        """
332        This method performs the following operations:
333        1. Logs the episode return if logging is enabled
334        2. Saves and empties the trace if save_path is specified
335        3. Resets all agents that have a reset method
336        4. Resets the lower level environment
337        5. Generates observation and info messages
338        6. Resets episode-related counters
339
340        Each returned element is a dict where the keys are the names of the higher level agents.
341
342        Returns:
343            Tuple[Dict[str, Dict[str, np.ndarray]], dict]: A tuple containing:
344                - msg_obs: Dictionary of observations for each agent
345                - msg_infos: Dictionary of info messages for each agent
346        """
347        if self.verbose >= 1:
348            print(f"{self.name} -  Reset")
349
350        # Log episode return
351        if self.episode_idx >= 0:
352            if self.logger is not None:
353                self.logger.add_scalar(
354                    f"{self.name}/total_return",
355                    np.sum([ret for ret in self.episode_returns.values()]),
356                    self.level_ts,
357                )
358            if self.verbose >= 2:
359                print(
360                    f"{self.name} -  level_step: {self.level_ts} - Ep. return={np.sum([ret for ret in self.episode_returns.values()])}"
361                )
362
363        # Save trace and empty it
364        if self.save_path is not None and self.trace is not None and len(self.trace):
365            self.trace.save_trace(save_path=self.save_path, episode=self.episode_idx)
366            self.trace.empty()
367
368        # Reset agents
369        for agent in self.agents.values():
370            if hasmethod(agent, "reset"):
371                agent.reset()  # type: ignore
372
373        # Reset lower level env
374        env_obs, env_inf = self.env.reset()
375        env_obs = self.save_obs_from_bottom(env_obs=env_obs)
376
377        msg_obs, _, _, _, msg_infos = self.make_message(
378            env_obs=env_obs,
379            env_rew=None,
380            env_term=None,
381            env_trunc=None,
382            env_inf=env_inf,
383        )
384
385        self.episode_returns = {ag: 0.0 for ag in self.agents}
386        self.episode_ts = 0
387        self.episode_idx += 1
388        return msg_obs, msg_infos

390    def save_obs_from_bottom(
391        self, env_obs: Dict[str, np.ndarray] | Dict[str, Dict[str, np.ndarray]]
392    ) -> Dict[str, Dict[str, np.ndarray]]:
393        """Wraps and stores the observation from the bottom level of the hierarchy.
394
395        This method takes observations from a lower level and reformats them if necessary to maintain
396        a consistent hierarchical structure. It supports both nested and flat observation dictionaries.
397
398        Args:
399            env_obs (Dict[str, np.ndarray] | Dict[str, Dict[str, np.ndarray]]): Observations from the lower level.
400                Can be either a flat dictionary mapping agent IDs to observations,
401                or a nested dictionary with agent-subagent structure.
402
403        Returns:
404            Dict[str, Dict[str, np.ndarray]]: A nested dictionary where:
405                - First level keys are agent IDs
406                - Second level keys are subagent IDs
407                - Values are the corresponding observations as numpy arrays
408
409        Note:
410            The method stores the processed observations in self.obs_from_bottom before returning them.
411        """
412        if is_nested_dict(env_obs):
413            self.obs_from_bottom = env_obs
414            return env_obs
415        else:
416            reformatted_obs = {}
417            for agent, subagents in self.downlinks.items():
418                reformatted_obs[agent] = {subag: env_obs[subag] for subag in subagents}
419            self.obs_from_bottom = reformatted_obs
420            return reformatted_obs

422    def make_agent_obs(
423        self,
424        action_from_top: Dict[str, np.ndarray] | None,
425        obs_from_bottom: Dict[str, Dict[str, np.ndarray]],
426    ) -> Dict[str, Dict[str, np.ndarray]]:
427        """Generates the observation for the agents on this level by using the observation from the bottom and
428        the actions from top
429
430        This function combines or replaces observations from lower level agents with actions from higher level
431        agents to create the observation space for the current level's agents.
432
433        Args:
434            action_from_top (Dict[str, np.ndarray] | None): Dictionary mapping agent names to their actions from
435                the higher level, or None if this is the top level
436            obs_from_bottom (Dict[str, Dict[str, np.ndarray]]): Dictionary of dictionaries containing observations
437                from lower level agents. First key is current level agent name, second key is lower level agent name
438
439        Returns:
440            Dict[str, Dict[str, np.ndarray]]: Dictionary of dictionaries containing the combined/replaced observations.
441                First key is current level agent name, second key is lower level agent name
442        """
443        if action_from_top is None:
444            return obs_from_bottom
445
446        agent_obs = {}  # It will have shape {this_level_agent: {bottom_agent: obs}}
447        if self.concat_obs:
448            for agent_name in action_from_top:  # These are the agents in this level
449                obs = {  # This combines the observation from bottom with the action from top
450                    sub_name: np.concat(
451                        [
452                            obs_from_bottom[agent_name][sub_name],
453                            np.atleast_1d(action_from_top[agent_name]),
454                        ]
455                    )
456                    for sub_name in obs_from_bottom[agent_name]
457                }
458                agent_obs[agent_name] = obs
459        else:
460            for agent_name in action_from_top:
461                obs = {  # This replaces the observation from bottom with the action from top
462                    sub_name: np.atleast_1d(action_from_top[agent_name])
463                    for sub_name in obs_from_bottom[agent_name]
464                }
465                agent_obs[agent_name] = obs
466        return agent_obs

468    def step(
469        self,
470        action: Dict[str, np.ndarray] | None,
471        training: bool = True,
472    ) -> Tuple[
473        Dict[str, Dict[str, np.ndarray]] | None,
474        Dict[str, float] | None,
475        Dict[str, bool] | None,
476        Dict[str, bool] | None,
477        dict | None,
478    ]:
479        """Perform a step in the hierarchical environment, processing actions from top to bottom and then bottom to top.
480
481        Args:
482            action (Dict[str, np.ndarray] | None): Actions from the higher level, containing observations for each agent at this level.
483                None if there are no actions from higher level.
484            training (bool, optional): Flag indicating whether the environment is in training mode. Defaults to True.
485
486        Returns:
487            Tuple containing five dictionaries:
488                Dict[str, Dict[str, np.ndarray]]: Observations for each agent
489                Dict[str, float]: Rewards for each agent
490                Dict[str, bool]: Termination flags for each agent
491                Dict[str, bool]: Truncation flags for each agent
492                dict: Additional information dictionary
493
494        Flow:
495            1. Creates agent observations using high-level actions and bottom-level observations
496            2. Executes actions at specified frequency (action_ifreq)
497            3. Steps through the environment with agent actions
498            4. Processes and stores transitions if in training mode
499            5. Updates traces and episode returns
500            6. Periodically saves agent models if save_path is specified
501
502        Note:
503            - The method handles both the downward (top to bottom) and upward (bottom to top) flow of information
504            - Training data is stored only when training=True
505            - Final observations are stored when episodes terminate"""
506
507        # Act
508        # ---------------------------
509        agent_obs = self.make_agent_obs(
510            action_from_top=action, obs_from_bottom=self.obs_from_bottom
511        )
512
513        # Act only every action K steps
514        # TODO this is quite ugly, the frequency should be decided by the agent not by the env
515        if self.episode_ts % self.action_ifreq == 0:
516            # Pass the higher level actions to the agents of this level as obs
517            self.env_actions = self.act(observation=agent_obs, training=training)
518
519        # Pass the actions of the agents of this level to the lower level env and get rew and ll_obs
520        if isinstance(self.env, LevelEnv):
521            env_obs, env_rew, env_term, env_trunc, env_inf = self.env.step(
522                self.env_actions, training=training
523            )
524            assert (
525                env_obs is not None
526            ), f"The level {self.env.name} should return to {self.name} valid observations"
527            assert (
528                env_term is not None
529            ), f"The level {self.env.name} should return to {self.name} valid terminations"
530            assert (
531                env_trunc is not None
532            ), f"The level {self.env.name} should return to {self.name} valid truncations"
533            assert (
534                env_rew is not None
535            ), f"The level {self.env.name} should return to {self.name} valid rewards"
536        else:
537            env_obs, env_rew, env_term, env_trunc, env_inf = self.env.step(
538                self.env_actions
539            )
540        env_obs = self.save_obs_from_bottom(env_obs=env_obs)
541
542        dones = {agent: (env_term[agent] or env_trunc[agent]) for agent in self.agents}
543        # ---------------------------
544
545        # Make message
546        # ---------------------------
547        msg_obs, msg_rew, msg_term, msg_trunc, msg_infos = self.make_message(
548            env_obs=env_obs,
549            env_rew=env_rew,
550            env_trunc=env_trunc,
551            env_term=env_term,
552            env_inf=env_inf,
553        )
554        # ---------------------------
555
556        # Train
557        # ---------------------------
558        last_agent_obs = None
559        # TODO probably should not store the transitions if no new action is done...
560        if training:
561            # No need for this when not training. (For now :P)
562            for agent_name, agent in self.agents.items():
563                # Get the actions from this agent
564                agent_actions = {
565                    subagent: self.env_actions[subagent]
566                    for subagent in self.downlinks[agent_name]
567                }
568
569                agent.store(
570                    state=agent_obs[agent_name],
571                    action=agent_actions,
572                    reward=env_rew[agent_name],
573                    done=dones[agent_name],
574                )
575
576                # NOTE
577                # We create a new agent_obs with the last state from the env and the same top action!
578                # we do this cause there will be no new action coming from top, as the top level will call the reset
579                # And then we store the last transition!
580                if dones[agent_name]:
581                    if last_agent_obs is None:
582                        last_agent_obs = self.make_agent_obs(
583                            action_from_top=action, obs_from_bottom=env_obs
584                        )
585
586                    agent.store(
587                        state=last_agent_obs[agent_name],
588                        action=None,
589                        reward=None,
590                        done=dones[agent_name],
591                    )
592
593            self.update_step()
594        # ---------------------------
595
596        # Save important infos
597        # ---------------------------
598        if self.trace is not None:
599            self.trace.add(
600                episode=self.episode_idx,
601                actions=self.env_actions,
602                observations=agent_obs,

399        Args:
400            env_obs (Dict[str, np.ndarray] | Dict[str, Dict[str, np.ndarray]]): Observations from the lower level.
401                Can be either a flat dictionary mapping agent IDs to observations,
402                or a nested dictionary with agent-subagent structure.

404        Returns:
405            Dict[str, Dict[str, np.ndarray]]: A nested dictionary where:
406                - First level keys are agent IDs
407                - Second level keys are subagent IDs
408                - Values are the corresponding observations as numpy arrays
409
410        Note:
411            The method stores the processed observations in self.obs_from_bottom before returning them.
412        """
413        if is_nested_dict(env_obs):
414            self.obs_from_bottom = env_obs
415            return env_obs
416        else:
417            reformatted_obs = {}
418            for agent, subagents in self.downlinks.items():
419                reformatted_obs[agent] = {subag: env_obs[subag] for subag in subagents}
420            self.obs_from_bottom = reformatted_obs

423    def make_agent_obs(
424        self,
425        action_from_top: Dict[str, np.ndarray] | None,
426        obs_from_bottom: Dict[str, Dict[str, np.ndarray]],
427    ) -> Dict[str, Dict[str, np.ndarray]]:
428        """Generates the observation for the agents on this level by using the observation from the bottom and
429        the actions from top
430
431        This function combines or replaces observations from lower level agents with actions from higher level
432        agents to create the observation space for the current level's agents.
433
434        Args:
435            action_from_top (Dict[str, np.ndarray] | None): Dictionary mapping agent names to their actions from
436                the higher level, or None if this is the top level
437            obs_from_bottom (Dict[str, Dict[str, np.ndarray]]): Dictionary of dictionaries containing observations
438                from lower level agents. First key is current level agent name, second key is lower level agent name
439
440        Returns:
441            Dict[str, Dict[str, np.ndarray]]: Dictionary of dictionaries containing the combined/replaced observations.
442                First key is current level agent name, second key is lower level agent name
443        """
444        if action_from_top is None:
445            return obs_from_bottom
446
447        agent_obs = {}  # It will have shape {this_level_agent: {bottom_agent: obs}}
448        if self.concat_obs:
449            for agent_name in action_from_top:  # These are the agents in this level
450                obs = {  # This combines the observation from bottom with the action from top
451                    sub_name: np.concat(
452                        [
453                            obs_from_bottom[agent_name][sub_name],
454                            np.atleast_1d(action_from_top[agent_name]),
455                        ]
456                    )
457                    for sub_name in obs_from_bottom[agent_name]
458                }
459                agent_obs[agent_name] = obs
460        else:
461            for agent_name in action_from_top:
462                obs = {  # This replaces the observation from bottom with the action from top
463                    sub_name: np.atleast_1d(action_from_top[agent_name])
464                    for sub_name in obs_from_bottom[agent_name]
465                }
466                agent_obs[agent_name] = obs
467        return agent_obs

469    def step(
470        self,
471        action: Dict[str, np.ndarray] | None,
472        training: bool = True,
473    ) -> Tuple[
474        Dict[str, Dict[str, np.ndarray]] | None,
475        Dict[str, float] | None,
476        Dict[str, bool] | None,
477        Dict[str, bool] | None,
478        dict | None,
479    ]:
480        """Perform a step in the hierarchical environment, processing actions from top to bottom and then bottom to top.
481
482        Args:
483            action (Dict[str, np.ndarray] | None): Actions from the higher level, containing observations for each agent at this level.
484                None if there are no actions from higher level.
485            training (bool, optional): Flag indicating whether the environment is in training mode. Defaults to True.
486
487        Returns:
488            Tuple containing five dictionaries:
489                Dict[str, Dict[str, np.ndarray]]: Observations for each agent
490                Dict[str, float]: Rewards for each agent
491                Dict[str, bool]: Termination flags for each agent
492                Dict[str, bool]: Truncation flags for each agent
493                dict: Additional information dictionary
494
495        Flow:
496            1. Creates agent observations using high-level actions and bottom-level observations
497            2. Executes actions at specified frequency (action_ifreq)
498            3. Steps through the environment with agent actions
499            4. Processes and stores transitions if in training mode
500            5. Updates traces and episode returns
501            6. Periodically saves agent models if save_path is specified
502
503        Note:
504            - The method handles both the downward (top to bottom) and upward (bottom to top) flow of information
505            - Training data is stored only when training=True
506            - Final observations are stored when episodes terminate"""
507
508        # Act
509        # ---------------------------
510        agent_obs = self.make_agent_obs(
511            action_from_top=action, obs_from_bottom=self.obs_from_bottom
512        )
513
514        # Act only every action K steps
515        # TODO this is quite ugly, the frequency should be decided by the agent not by the env
516        if self.episode_ts % self.action_ifreq == 0:
517            # Pass the higher level actions to the agents of this level as obs
518            self.env_actions = self.act(observation=agent_obs, training=training)
519
520        # Pass the actions of the agents of this level to the lower level env and get rew and ll_obs
521        if isinstance(self.env, LevelEnv):
522            env_obs, env_rew, env_term, env_trunc, env_inf = self.env.step(
523                self.env_actions, training=training
524            )
525            assert (
526                env_obs is not None
527            ), f"The level {self.env.name} should return to {self.name} valid observations"
528            assert (
529                env_term is not None
530            ), f"The level {self.env.name} should return to {self.name} valid terminations"
531            assert (
532                env_trunc is not None
533            ), f"The level {self.env.name} should return to {self.name} valid truncations"
534            assert (
535                env_rew is not None
536            ), f"The level {self.env.name} should return to {self.name} valid rewards"
537        else:
538            env_obs, env_rew, env_term, env_trunc, env_inf = self.env.step(
539                self.env_actions
540            )
541        env_obs = self.save_obs_from_bottom(env_obs=env_obs)
542
543        dones = {agent: (env_term[agent] or env_trunc[agent]) for agent in self.agents}
544        # ---------------------------
545
546        # Make message
547        # ---------------------------
548        msg_obs, msg_rew, msg_term, msg_trunc, msg_infos = self.make_message(
549            env_obs=env_obs,
550            env_rew=env_rew,
551            env_trunc=env_trunc,
552            env_term=env_term,
553            env_inf=env_inf,
554        )
555        # ---------------------------
556
557        # Train
558        # ---------------------------
559        last_agent_obs = None
560        # TODO probably should not store the transitions if no new action is done...
561        if training:
562            # No need for this when not training. (For now :P)
563            for agent_name, agent in self.agents.items():
564                # Get the actions from this agent
565                agent_actions = {
566                    subagent: self.env_actions[subagent]
567                    for subagent in self.downlinks[agent_name]
568                }
569
570                agent.store(
571                    state=agent_obs[agent_name],
572                    action=agent_actions,
573                    reward=env_rew[agent_name],
574                    done=dones[agent_name],
575                )
576
577                # NOTE
578                # We create a new agent_obs with the last state from the env and the same top action!
579                # we do this cause there will be no new action coming from top, as the top level will call the reset
580                # And then we store the last transition!
581                if dones[agent_name]:
582                    if last_agent_obs is None:
583                        last_agent_obs = self.make_agent_obs(
584                            action_from_top=action, obs_from_bottom=env_obs
585                        )
586
587                    agent.store(
588                        state=last_agent_obs[agent_name],
589                        action=None,
590                        reward=None,
591                        done=dones[agent_name],
592                    )
593
594            self.update_step()
595        # ---------------------------
596
597        # Save important infos
598        # ---------------------------
599        if self.trace is not None:
600            self.trace.add(
601                episode=self.episode_idx,
602                actions=self.env_actions,
603                observations=agent_obs,
604                rewards=env_rew,
605                terminations=env_term,
606                truncations=env_trunc,
607                # These are the ones sent to the higher level
608                return_observations=msg_obs,
609                return_rewards=msg_rew,
610            )
611
612            if last_agent_obs is not None:
613                self.trace.add_final_obs(last_agent_obs)
614
615        for agent in self.episode_returns:
616            self.episode_returns[agent] += env_rew[agent]
617
618        if self.level_ts % 100000 == 0 and self.save_path is not None:
619            if self.verbose >= 2:
620                print(
621                    f"{self.name} -  Saving model at level step {self.level_ts} in {self.save_path}"
622                )
623            with open(self.save_path / "last_model.json", "w") as f:
624                json.dump({"save_step": self.level_ts}, f)
625            for agent_name, agent in self.agents.items():
626                agent.save_agent(save_path=self.save_path, name=agent_name)
627        # ---------------------------
628
629        self.level_ts += 1
630        self.episode_ts += 1
631        return msg_obs, msg_rew, msg_term, msg_trunc, msg_infos
