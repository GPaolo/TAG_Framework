148    def add_level_config(self, level_cfg: LevelConfig) -> None:
149        """Adds a level to the hierarchy based on the provided configuration.
150
151        This method creates a new level in the hierarchy by instantiating agents according to the level
152        configuration and connecting them with the previous level. It handles observation space modifications
153        when concatenating observations from higher levels.
154
155        Args:
156            level_cfg (LevelConfig): Configuration object containing all necessary parameters to create
157                a new level, including:
158                - env: Environment for this level
159                - agents: List of agent configurations
160                - name: Level name
161                - concat_obs: Whether to concatenate observations from higher levels
162                - action_space: Space of actions from higher levels (required if concat_obs=True)
163                - action_frequency: Frequency of actions
164                - trace_type: Type of trace to use
165                - uplinks/downlinks: Communication links between levels
166
167        Raises:
168            ValueError: If concat_obs is True but no action_space is provided
169            TypeError: If observation spaces are incompatible when merging spaces
170
171        Notes:
172            - If concat_obs is True, the observation space of each agent is extended to include
173              actions from higher levels
174            - The first level added becomes the base environment of the hierarchy
175        """
176        prev_level = level_cfg.env
177
178        # Create agents
179        agents = {}
180        for cfg in level_cfg.agents:
181            # Find action space from top
182            # ---------------------------
183            agent_additional_obs_space = None
184            if level_cfg.concat_obs:
185                if level_cfg.action_space is None:
186                    raise ValueError(
187                        f"Must provide an action space for the level {level_cfg.name} if concat_obs is True"
188                    )
189
190                for upper_agent in level_cfg.action_space:
191                    if cfg.name in level_cfg.action_space[upper_agent].keys():  # type: ignore
192                        agent_additional_obs_space = level_cfg.action_space[
193                            upper_agent
194                        ][cfg.name]  # type: ignore
195                        break
196            # ---------------------------
197
198            # Make new observation space for agent
199            # We take observation space of the env and add the action size from the top
200            # ---------------------------
201            # TODO make sure that we can have different names for the agents in the bottom level
202            # TODO so, we should handle the fact that the observation space from the env is not a nested dict!
203            # TODO so something that could be done is to have: if prev_lev not LevelEnv: wrap spaces...
204            if isinstance(agent_additional_obs_space, Discrete):
205                if isinstance(prev_level.observation_space(cfg.name), Box):
206                    agent_obs_space = extend_box_space(
207                        original_space=prev_level.observation_space(cfg.name),  # type: ignore
208                        add_size=1,
209                    )
210                elif isinstance(prev_level.observation_space(cfg.name), GymDict):
211                    agent_obs_space = extend_dict_box_space(
212                        original_space=prev_level.observation_space(cfg.name),  # type: ignore
213                        add_size=1,
214                    )
215                else:
216                    raise TypeError(
217                        f"For level {level_cfg.name} and agent {cfg.name} previous level observation space must be Box or GymDict[str: Box]."
218                    )
219            elif isinstance(agent_additional_obs_space, Box):
220                if isinstance(prev_level.observation_space(cfg.name), Box):
221                    agent_obs_space = merge_spaces(
222                        [
223                            prev_level.observation_space(cfg.name),  # type: ignore
224                            agent_additional_obs_space,
225                        ]
226                    )
227                else:
228                    raise TypeError(
229                        f"For level {level_cfg.name} and agent {cfg.name} both observation spaces must be of type Box to merge."
230                    )
231            else:
232                agent_obs_space = prev_level.observation_space(cfg.name)
233
234            if not isinstance(agent_obs_space, GymDict):
235                agent_obs_space = GymDict({cfg.name: agent_obs_space})
236
237            agent_action_space = prev_level.action_space(cfg.name)
238            if not isinstance(agent_action_space, GymDict):
239                agent_action_space = GymDict({cfg.name: agent_action_space})
240            # ---------------------------
241
242            # Instantiate agent
243            agents[cfg.name] = cfg.agent_class(
244                name=cfg.name,
245                observation_space=agent_obs_space,
246                action_space=agent_action_space,
247                communication_space=cfg.communication_space,
248                device=cfg.device,
249                args=cfg.agent_kwargs.get("args", None),
250            )
251
252        # Create level
253        level = LevelEnv(
254            agents=agents,
255            uplinks=level_cfg.uplinks,
256            downlinks=level_cfg.downlinks,
257            env=prev_level,
258            name=level_cfg.name,
259            action_ifreq=level_cfg.action_frequency,
260            trace_type=level_cfg.trace_type,
261            concat_obs=level_cfg.concat_obs,
262            action_space=level_cfg.action_space,
263        )
264
265        if len(self.levels) == 0:
266            self.env = level_cfg.env
267        self.add_level(level)

342    def act(self, observation: Dict[str, Any]) -> Dict[str, Any]:
343        """
344        Allow observations from environment to travel up the hierarchy and actions to travel down.
345
346        This method implements a two-pass hierarchical process:
347        1. Bottom-up pass: Observations travel up the hierarchy, with each level generating messages
348            for the level above it.
349        2. Top-down pass: Actions are generated at each level and passed down the hierarchy.
350
351        Args:
352             observation (Dict[str, Any]): Environment observation dictionary mapping agent names to
353                  their observations.
354
355        Returns:
356             Dict[str, Any]: Dictionary of actions for the environment mapping agent names to their
357                  corresponding actions.
358
359        Flow:
360             1. Bottom-up:
361                  - Each level processes observations from below
362                  - Generates messages to pass upwards
363                  - Messages become observations for next level
364             2. Top-down:
365                  - Starting from top level
366                  - Each level receives action from above
367                  - Combines with bottom-up message to generate new action
368                  - Passes action down to next level
369        """
370        # First propagate observations up the hierarchy from bottom to top
371        obs = observation
372
373        # Bottom-up pass: Collect messages from each level
374        for level in self.levels:
375            # Save and format observation for this level
376            prepared_obs = level.save_obs_from_bottom(obs) # type: ignore
377
378            # Generate message to send up
379            msg = level.make_message(
380                env_obs=prepared_obs,
381                env_rew=None,
382                env_term=None,
383                env_trunc=None,
384                env_inf=None,
385            )[0]  # Take first element since make_message returns tuple
386
387            obs = msg  # Pass message as observation to next level
388
389        # Top-down pass: Generate actions at each level
390        action = None
391
392        # Iterate through levels in reverse, from top to bottom
393        for i in range(len(self.levels) - 1, -1, -1):
394            level = self.levels[i]
395
396            # Create observation for this level by combining bottom-up message
397            # with top-down action
398            level_obs = level.make_agent_obs(
399                action_from_top=action, obs_from_bottom=level.obs_from_bottom
400            )
401
402            # Get action from this level to pass down
403            action = level.act(observation=level_obs)
404
405        return action  # type: ignore

407    def tree(self):
408        """Returns a hierarchical tree representation of the associations between levels.
409
410        This method constructs a nested dictionary representing the hierarchical structure,
411        starting from the lowest level and building up through the uplinks between levels.
412        Each level's associations are merged into a tree structure where keys are nodes
413        and values are their children/descendants.
414
415        Returns:
416            dict: A nested dictionary representing the hierarchical tree structure.
417                  The keys are parent nodes and values are dictionaries containing
418                  their children and further descendants.
419
420        Example:
421            If Level 0 has nodes [1,2,3] and Level 1 has nodes [A,B] with
422            A linked to [1,2] and B linked to [3], the result would be:
423            {'A': {'1': {}, '2': {}}, 'B': {'3': {}}}
424        """
425        tree = self.levels[0].downlinks
426        for level in self.levels:
427            if level.uplinks is None:
428                break
429
430            local_tree = {}
431            for up, down in level.uplinks.items():
432                local_tree[up] = {d: tree[d] for d in down}
433            tree = local_tree
434        return tree

436    def print_tree(self, tree: None | dict = None, indent: int = 0, prefix: str = ""):
437        """
438        Prints a hierarchical tree representation of the agent hierarchy.
439
440        This method visualizes the hierarchical structure using ASCII characters, where leaf nodes
441        (environments) are marked with '(env)' and intermediate nodes represent higher-level agents.
442
443        Args:
444            tree (dict | None, optional): The hierarchical tree structure to print. If None,
445                uses the tree from self.tree(). Defaults to None.
446            indent (int, optional): The current indentation level. Used for recursive calls.
447                Defaults to 0.
448            prefix (str, optional): The prefix string for the current line. Used for recursive
449                calls to maintain the tree structure. Defaults to "".
450
451        Example:
452            For a hierarchy with two high-level agents and their environments:
453            Agent1
454            ├── env1 (env)
455            └── env2 (env)
456            Agent2
457            └── env3 (env)
458        """
459        if tree is None:
460            tree = self.tree()
461
462        items = list(tree.items())
463        for i, (key, value) in enumerate(items):
464            is_last = i == len(items) - 1
465            current_prefix = "└── " if is_last else "├── "
466            # Higher level agents (with children) don't get env flag
467            print(prefix + current_prefix + str(key))
468
469            new_prefix = prefix + ("    " if is_last else "│   ")
470            if isinstance(value, dict):
471                self.print_tree(value, indent + 1, new_prefix)
472            elif isinstance(value, list):
473                for j, item in enumerate(value):
474                    is_last_item = j == len(value) - 1
475                    item_prefix = "└── " if is_last_item else "├── "
476                    # Add (env) flag to leaf nodes
477                    print(new_prefix + item_prefix + str(item) + " (env)")

479    def print_hierarchy_details(self):
480        """Print detailed information about hierarchy structure, including level and agent spaces.
481
482        This method provides a comprehensive visualization of the hierarchy's structure, printing
483        information about each level including:
484        - Level name and index
485        - Level-wide observation and action spaces
486        - Uplink and downlink connectivity between agents
487        - Per-agent details:
488            - Observation space
489            - Action space
490            - Communication space
491            - Input connections from lower-level agents (for non-bottom levels)
492
493        The output is formatted with clear section headers and separators for readability.
494
495        Returns:
496            None
497        """
498        print("\n=== Hierarchy Details ===\n")
499
500        for level_idx, level in enumerate(self.levels):
501            print(f"Level {level_idx}: {level.name}")
502            print("─" * 60)
503
504            # Print level spaces
505            print("\nLevel Spaces:")
506            print(f"  Observation Space: {level.observation_spaces}")
507            print(f"  Action Space: {level.action_spaces}")
508
509            # Print connectivity
510            if level.uplinks:
511                print("\nUplinks:")
512                for up, down in level.uplinks.items():
513                    print(f"  {up} → {down}")
514
515            if level.downlinks:
516                print("\nDownlinks:")
517                for agent, children in level.downlinks.items():
518                    print(f"  {agent} → {children}")
519
520            # Print agent details
521            print("\nAgents:")
522            for agent_name, agent in level.agents.items():
523                print(f"  {agent_name}:")
524                print(f"    Observation space: {agent.observation_space}")
525                print(f"    Action space: {agent.action_space}")
526                print(f"    Communication space: {agent.communication_space}")
527
528                # Show space transformations if applicable
529                if level_idx > 0:  # Not bottom level
530                    print("    Receives from below:")
531                    for child in level.downlinks.get(agent_name, []):
532                        child_agent = self.levels[level_idx - 1].agents[child]
533                        print(f"      {child}: {child_agent.communication_space}")
534
535            print("\n" + "=" * 60 + "\n")

153        configuration and connecting them with the previous level. It handles observation space modifications
154        when concatenating observations from higher levels.
155
156        Args:
157            level_cfg (LevelConfig): Configuration object containing all necessary parameters to create
158                a new level, including:
159                - env: Environment for this level

161                - name: Level name
162                - concat_obs: Whether to concatenate observations from higher levels
163                - action_space: Space of actions from higher levels (required if concat_obs=True)
164                - action_frequency: Frequency of actions
165                - trace_type: Type of trace to use
166                - uplinks/downlinks: Communication links between levels
167

169            ValueError: If concat_obs is True but no action_space is provided
170            TypeError: If observation spaces are incompatible when merging spaces
171
172        Notes:
173            - If concat_obs is True, the observation space of each agent is extended to include
174              actions from higher levels
175            - The first level added becomes the base environment of the hierarchy
176        """
177        prev_level = level_cfg.env
178
179        # Create agents
180        agents = {}
181        for cfg in level_cfg.agents:
182            # Find action space from top
183            # ---------------------------
184            agent_additional_obs_space = None
185            if level_cfg.concat_obs:
186                if level_cfg.action_space is None:
187                    raise ValueError(
188                        f"Must provide an action space for the level {level_cfg.name} if concat_obs is True"
189                    )
190
191                for upper_agent in level_cfg.action_space:
192                    if cfg.name in level_cfg.action_space[upper_agent].keys():  # type: ignore
193                        agent_additional_obs_space = level_cfg.action_space[
194                            upper_agent
195                        ][cfg.name]  # type: ignore
196                        break
197            # ---------------------------
198
199            # Make new observation space for agent
200            # We take observation space of the env and add the action size from the top
201            # ---------------------------
202            # TODO make sure that we can have different names for the agents in the bottom level
203            # TODO so, we should handle the fact that the observation space from the env is not a nested dict!
204            # TODO so something that could be done is to have: if prev_lev not LevelEnv: wrap spaces...
205            if isinstance(agent_additional_obs_space, Discrete):
206                if isinstance(prev_level.observation_space(cfg.name), Box):

208                        original_space=prev_level.observation_space(cfg.name),  # type: ignore

219                    )
220            elif isinstance(agent_additional_obs_space, Box):
221                if isinstance(prev_level.observation_space(cfg.name), Box):
222                    agent_obs_space = merge_spaces(

343    def act(self, observation: Dict[str, Any]) -> Dict[str, Any]:
344        """
345        Allow observations from environment to travel up the hierarchy and actions to travel down.
346
347        This method implements a two-pass hierarchical process:
348        1. Bottom-up pass: Observations travel up the hierarchy, with each level generating messages
349            for the level above it.
350        2. Top-down pass: Actions are generated at each level and passed down the hierarchy.
351
352        Args:
353             observation (Dict[str, Any]): Environment observation dictionary mapping agent names to
354                  their observations.
355
356        Returns:
357             Dict[str, Any]: Dictionary of actions for the environment mapping agent names to their
358                  corresponding actions.
359
360        Flow:
361             1. Bottom-up:
362                  - Each level processes observations from below
363                  - Generates messages to pass upwards
364                  - Messages become observations for next level
365             2. Top-down:
366                  - Starting from top level
367                  - Each level receives action from above
368                  - Combines with bottom-up message to generate new action
369                  - Passes action down to next level
370        """
371        # First propagate observations up the hierarchy from bottom to top
372        obs = observation
373
374        # Bottom-up pass: Collect messages from each level
375        for level in self.levels:
376            # Save and format observation for this level
377            prepared_obs = level.save_obs_from_bottom(obs) # type: ignore
378
379            # Generate message to send up
380            msg = level.make_message(
381                env_obs=prepared_obs,
382                env_rew=None,
383                env_term=None,
384                env_trunc=None,
385                env_inf=None,
386            )[0]  # Take first element since make_message returns tuple
387
388            obs = msg  # Pass message as observation to next level
389
390        # Top-down pass: Generate actions at each level
391        action = None
392
393        # Iterate through levels in reverse, from top to bottom
394        for i in range(len(self.levels) - 1, -1, -1):
395            level = self.levels[i]
396
397            # Create observation for this level by combining bottom-up message
398            # with top-down action
399            level_obs = level.make_agent_obs(
400                action_from_top=action, obs_from_bottom=level.obs_from_bottom
401            )
402
403            # Get action from this level to pass down
404            action = level.act(observation=level_obs)
405
406        return action  # type: ignore

408    def tree(self):
409        """Returns a hierarchical tree representation of the associations between levels.
410
411        This method constructs a nested dictionary representing the hierarchical structure,
412        starting from the lowest level and building up through the uplinks between levels.
413        Each level's associations are merged into a tree structure where keys are nodes
414        and values are their children/descendants.
415
416        Returns:
417            dict: A nested dictionary representing the hierarchical tree structure.
418                  The keys are parent nodes and values are dictionaries containing
419                  their children and further descendants.
420
421        Example:
422            If Level 0 has nodes [1,2,3] and Level 1 has nodes [A,B] with
423            A linked to [1,2] and B linked to [3], the result would be:
424            {'A': {'1': {}, '2': {}}, 'B': {'3': {}}}
425        """
426        tree = self.levels[0].downlinks
427        for level in self.levels:
428            if level.uplinks is None:
429                break
430
431            local_tree = {}
432            for up, down in level.uplinks.items():
433                local_tree[up] = {d: tree[d] for d in down}
434            tree = local_tree
435        return tree

437    def print_tree(self, tree: None | dict = None, indent: int = 0, prefix: str = ""):
438        """
439        Prints a hierarchical tree representation of the agent hierarchy.
440
441        This method visualizes the hierarchical structure using ASCII characters, where leaf nodes
442        (environments) are marked with '(env)' and intermediate nodes represent higher-level agents.
443
444        Args:
445            tree (dict | None, optional): The hierarchical tree structure to print. If None,
446                uses the tree from self.tree(). Defaults to None.
447            indent (int, optional): The current indentation level. Used for recursive calls.
448                Defaults to 0.
449            prefix (str, optional): The prefix string for the current line. Used for recursive
450                calls to maintain the tree structure. Defaults to "".
451
452        Example:
453            For a hierarchy with two high-level agents and their environments:
454            Agent1
455            ├── env1 (env)
456            └── env2 (env)
457            Agent2
458            └── env3 (env)
459        """
460        if tree is None:
461            tree = self.tree()
462
463        items = list(tree.items())
464        for i, (key, value) in enumerate(items):
465            is_last = i == len(items) - 1
466            current_prefix = "└── " if is_last else "├── "
467            # Higher level agents (with children) don't get env flag
468            print(prefix + current_prefix + str(key))
469
470            new_prefix = prefix + ("    " if is_last else "│   ")
471            if isinstance(value, dict):
472                self.print_tree(value, indent + 1, new_prefix)
473            elif isinstance(value, list):
474                for j, item in enumerate(value):
475                    is_last_item = j == len(value) - 1
476                    item_prefix = "└── " if is_last_item else "├── "
477                    # Add (env) flag to leaf nodes
478                    print(new_prefix + item_prefix + str(item) + " (env)")

343    def act(self, observation: Dict[str, Any]) -> Dict[str, Any]:
344        """
345        Allow observations from environment to travel up the hierarchy and actions to travel down.
346
347        This method implements a two-pass hierarchical process:
348        1. Bottom-up pass: Observations travel up the hierarchy, with each level generating messages
349            for the level above it.
350        2. Top-down pass: Actions are generated at each level and passed down the hierarchy.
351
352        Args:
353             observation (Dict[str, Any]): Environment observation dictionary mapping agent names to
354                  their observations.
355
356        Returns:
357             Dict[str, Any]: Dictionary of actions for the environment mapping agent names to their
358                  corresponding actions.
359
360        Flow:
361             1. Bottom-up:
362                  - Each level processes observations from below
363                  - Generates messages to pass upwards
364                  - Messages become observations for next level
365             2. Top-down:
366                  - Starting from top level
367                  - Each level receives action from above
368                  - Combines with bottom-up message to generate new action
369                  - Passes action down to next level
370        """
371        # First propagate observations up the hierarchy from bottom to top
372        obs = observation
373
374        # Bottom-up pass: Collect messages from each level
375        for level in self.levels:
376            # Save and format observation for this level
377            prepared_obs = level.save_obs_from_bottom(obs) # type: ignore
378
379            # Generate message to send up
380            msg = level.make_message(
381                env_obs=prepared_obs,
382                env_rew=None,
383                env_term=None,
384                env_trunc=None,
385                env_inf=None,
386            )[0]  # Take first element since make_message returns tuple
387
388            obs = msg  # Pass message as observation to next level
389
390        # Top-down pass: Generate actions at each level
391        action = None
392
393        # Iterate through levels in reverse, from top to bottom
394        for i in range(len(self.levels) - 1, -1, -1):
395            level = self.levels[i]
396
397            # Create observation for this level by combining bottom-up message
398            # with top-down action
399            level_obs = level.make_agent_obs(
400                action_from_top=action, obs_from_bottom=level.obs_from_bottom
401            )
402
403            # Get action from this level to pass down
404            action = level.act(observation=level_obs)
405
406        return action  # type: ignore

def tree(self): 
408    def tree(self):
409        """Returns a hierarchical tree representation of the associations between levels.
410
411        This method constructs a nested dictionary representing the hierarchical structure,
412        starting from the lowest level and building up through the uplinks between levels.
413        Each level's associations are merged into a tree structure where keys are nodes
414        and values are their children/descendants.
415
416        Returns:
417            dict: A nested dictionary representing the hierarchical tree structure.
418                  The keys are parent nodes and values are dictionaries containing
419                  their children and further descendants.
420
421        Example:
422            If Level 0 has nodes [1,2,3] and Level 1 has nodes [A,B] with
423            A linked to [1,2] and B linked to [3], the result would be:
424            {'A': {'1': {}, '2': {}}, 'B': {'3': {}}}
425        """
426        tree = self.levels[0].downlinks
427        for level in self.levels:
428            if level.uplinks is None:
429                break
430
431            local_tree = {}
432            for up, down in level.uplinks.items():
433                local_tree[up] = {d: tree[d] for d in down}
434            tree = local_tree
435        return tree

438        """
439        Prints a hierarchical tree representation of the agent hierarchy.
440
441        This method visualizes the hierarchical structure using ASCII characters, where leaf nodes
442        (environments) are marked with '(env)' and intermediate nodes represent higher-level agents.
443
444        Args:
445            tree (dict | None, optional): The hierarchical tree structure to print. If None,

453            For a hierarchy with two high-level agents and their environments:

466            current_prefix = "└── " if is_last else "├── "
467            # Higher level agents (with children) don't get env flag

469
470            new_prefix = prefix + ("    " if is_last else "│   ")

477                    # Add (env) flag to leaf nodes
