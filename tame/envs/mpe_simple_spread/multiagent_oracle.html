 68class GoalAssigner:
 69    """A class to assign goals to multiple agents in a coordinated manner.
 70
 71    This class implements a high-level coordinator that assigns target goals to each subagent
 72    in a multi-agent system. It uses the Hungarian algorithm (linear sum assignment) to optimally
 73    match agents to goals based on distances.
 74
 75    Args:
 76        n_subagent (int): Number of subagents/robots in the system.
 77
 78    Methods:
 79        assign_goals(observation: Dict[str, np.ndarray]) -> dict:
 80            Assigns goals to each agent by minimizing the total distance between agents and goals.
 81
 82            Args:
 83                observation (Dict[str, np.ndarray]): Dictionary containing observations for each agent.
 84                    Each observation should contain the agent's position and goal positions.
 85
 86            Returns:
 87                dict: Dictionary mapping agent names to their assigned goal positions.
 88                    Format: {agent_name: np.ndarray([x, y])}
 89
 90    Note:
 91        The observation dictionary for each agent should contain:
 92        - Agent position at indices [2:4]
 93        - Goal positions starting at index 4
 94    """
 95
 96    def __init__(self, n_subagent: int) -> None:
 97        self.n_subagents = n_subagent
 98
 99    def assign_goals(self, observation: Dict[str, np.ndarray]) -> dict:
100        """Assigns goals to agents based on their positions using the Hungarian algorithm.
101
102        This method takes the current observation of all agents and assigns goals to minimize
103        the total distance between agents and their assigned goals.
104
105        Args:
106            observation (Dict[str, np.ndarray]): A dictionary containing observations for each agent.
107                Each observation includes the agent's position and the relative positions of all goals.
108
109        Returns:
110            dict: A dictionary mapping agent names to their assigned absolute goal positions.
111                The goals are assigned to minimize the sum of distances between agents and goals.
112
113        Note:
114            The observation structure for each agent contains:
115            - Agent position at indices [2:4]
116            - Relative goal positions starting at index 4
117            Goals are converted from relative to absolute positions before assignment.
118        """
119        # Get absolute goals positions
120        goal_poses = None
121        agent_poses = []
122        for agent_name in observation:
123            feedback = observation[agent_name]
124            agent_pose = feedback[2:4]
125            if goal_poses is None:
126                limit = 4 + self.n_subagents * 2
127                goal_poses = feedback[4:limit].reshape(-1, 2) + agent_pose
128            agent_poses.append(agent_pose)
129        agent_poses = np.array(agent_poses)
130
131        # Assign goals
132        distance_matrix = np.linalg.norm(
133            agent_poses[:, np.newaxis] - goal_poses, axis=2
134        )
135        _, goals_idx = linear_sum_assignment(distance_matrix)
136        goals = {}
137        for goal_idx, agent_name in zip(goals_idx, observation):
138            goals[agent_name] = goal_poses[goal_idx]  # type: ignore
139        return goals

142class SubAgent:
143    """A class representing a sub-agent that can navigate towards a goal position.
144
145    The SubAgent implements simple goal-directed navigation behavior using a heuristic
146    approach based on comparing x and y distances to the goal.
147
148    Attributes:
149        goal (numpy.ndarray): Target position the agent should move towards.
150        tolerance (float): Minimum distance threshold before considering a direction change.
151
152    Methods:
153        give_goal(goal: numpy.ndarray): Sets the target position for the agent.
154        act(observation: numpy.ndarray) -> int: Determines movement action based on current position
155            and goal.
156
157    Returns:
158        int: Action index where:
159            0 = no movement
160            1 = move left
161            2 = move right
162            3 = move down
163            4 = move up
164    """
165
166    def __init__(self):
167        self.goal = None
168        self.tolerance = 0.0
169
170    def give_goal(self, goal: np.ndarray):
171        """
172        Sets the goal state for the multiagent oracle.
173
174        Args:
175            goal (np.ndarray): The target state/position to be reached by the agents.
176
177        Returns:
178            None
179
180        Note:
181            The goal state is stored as an instance variable and will be used for subsequent
182            calculations or decision-making processes.
183        """
184        self.goal = goal
185
186    def act(self, observation: np.ndarray) -> int:
187        """Move the agent towards a predefined goal using a heuristic approach.
188
189        The function determines the next action by comparing the relative position
190        between the agent's current position and its goal, prioritizing movement
191        along the axis with the larger difference.
192
193        Args:
194            observation (np.ndarray): The current observation of the environment,
195                where indices [2:4] contain the agent's current position.
196
197        Returns:
198            int: The action to take:
199                0 - no movement
200                1 - move left
201                2 - move right
202                3 - move down
203                4 - move up
204
205        Note:
206            The function uses a tolerance parameter to prevent oscillations
207            when the agent is close to the goal.
208        """
209        pose = observation[2:4]
210        direction_to_goal = self.goal - pose
211
212        # Select action based on direction (this is a heuristic approach)
213        if abs(direction_to_goal[0]) > abs(direction_to_goal[1]) + self.tolerance:
214            if direction_to_goal[0] > 0 + self.tolerance:
215                return 2  # move_right
216            elif direction_to_goal[0] < 0 - self.tolerance:
217                return 1  # move_left
218        elif abs(direction_to_goal[0]) < abs(direction_to_goal[1]):
219            if direction_to_goal[1] > 0 + self.tolerance:
220                return 4  # move_up
221            elif direction_to_goal[1] < 0 - self.tolerance:
222                return 3  # move_down
223        return 0

226class Agent(BaseAgent):
227    """A hierarchical agent class that implements an Hand-designed heuristic for the MPE spread environemnt.
228
229    This agent class implements a two-level hierarchical control structure where:
230    1. A high-level goal assigner allocates goals to individual agents
231    2. Low-level agents execute actions to reach their assigned goals
232
233    The agent supports training in parallel environments and handles episode tracking,
234    logging, and visualization through tensorboard.
235
236    Attributes:
237        args (Args): Configuration arguments for the agent
238        device (torch.device): Device to run computations on (CPU/GPU)
239        env (ParallelEnv): The multi-agent environment
240        num_env_agents (int): Number of agents in the environment
241        LL_agents (List[SubAgent]): List of low-level agents
242        goal_assigner (GoalAssigner): The high-level goal assignment module
243        assigned_goals (Optional[Dict]): Current goal assignments for agents
244        trained (bool): Whether the agent has completed training
245
246    Methods:
247        seed(): Sets random seeds for reproducibility
248        reset(): Resets the agent's internal state
249        train(): Trains the agent in the environment
250        act(): Generates actions for all agents based on observations
251        load_agent(): Loads a saved agent state
252        save_agent(): Saves the current agent state
253    """
254
255    def __init__(self, env: ParallelEnv, args: None | Args = None) -> None:
256        if args is None:
257            self.args: Args = Args()
258        else:
259            self.args: Args = args
260
261        if self.args.cuda >= 0:
262            self.device = torch.device(
263                f"cuda:{self.args.cuda}" if torch.cuda.is_available() else "cpu"
264            )
265        self.seed()
266
267        # env setup
268        self.env = env
269        self.num_env_agents = len(self.env.possible_agents)
270
271        self.LL_agents: List[SubAgent] = [
272            SubAgent() for _ in range(self.num_env_agents)
273        ]
274        self.goal_assigner = GoalAssigner(self.num_env_agents)
275        self.assigned_goals = None
276
277    def seed(self):
278        """Sets random seeds for reproducibility across different random number generators.
279
280        This method initializes the random seeds for the Python random module, NumPy,
281        and PyTorch to ensure reproducible results across runs. It also configures PyTorch's
282        CUDNN backend determinism if specified in arguments.
283
284        Args:
285            None
286
287        Returns:
288            None
289
290        Note:
291            The method uses the seed value stored in self.args.seed for all random generators
292            and sets torch.backends.cudnn.deterministic based on self.args.torch_deterministic
293        """
294        # TRY NOT TO MODIFY: seeding
295        random.seed(self.args.seed)
296        np.random.seed(self.args.seed)
297        torch.manual_seed(self.args.seed)
298        torch.backends.cudnn.deterministic = self.args.torch_deterministic

300    def reset(self):
301        """
302        Reset the MultiagentOracle environment.
303
304        This method resets the goal assignments for the agents in the environment. After
305        calling this method, no agent will have an assigned goal until new assignments
306        are made.
307
308        Returns:
309            None
310        """
311        self.assigned_goals = None

313    def train(
314        self, env: Any, log_path: Path | str | None = None, run_name: str | None = None
315    ):
316        """Runs the oracle on the provided environment.
317
318        This method implements the training loop for the oracle, collecting experience,
319        logging metrics, and saving traces of the interactions.
320        In practice there is no learning.
321        The oracle coordinates multiple agents to achieve their goals in the environment by following hand-designed heuristic.
322
323        Basically each subagent is assigned a goal in the environment and it moves directly towards it.
324
325        Args:
326            env (Any): The environment to train on. Must implement gym-like interface.
327            log_path (Path | str | None, optional): Directory path to save logs and traces.
328                Defaults to "runs" if None.
329            run_name (str | None, optional): Name for this training run. If None, generates
330                name using experiment name, seed and timestamp.
331
332        Returns:
333            None: The method updates the oracle's internal state and saves logs/traces to disk.
334
335        The training process:
336        1. Sets up logging directories and saves hyperparameters
337        2. Runs episodes collecting experience until total_timesteps is reached
338        3. For each step:
339           - Gets actions from the oracle
340           - Executes environment steps
341           - Tracks returns and done conditions
342           - Logs metrics and saves traces
343        4. Marks oracle as trained upon completion
344
345        Logs include:
346        - Per-agent episode returns
347        - Total episode returns
348        - Hyperparameters
349        - Full interaction traces if save_all_trace=True
350        """
351        if run_name is None:
352            run_name = f"{self.args.exp_name}__{self.args.seed}__{int(time.time())}"
353
354        if log_path is None:
355            log_path = Path("runs")
356        else:
357            log_path = Path(log_path)
358
359        args_dict = self.args.to_dict()  # type: ignore
360        save_path = log_path / run_name / "training"
361        os.makedirs(save_path, exist_ok=True)
362
363        with open(save_path / "params.json", "w") as f:
364            json.dump(args_dict, f, indent=4)
365
366        writer = SummaryWriter(save_path / "tboard")
367        writer.add_text(
368            "hyperparameters",
369            "|param|value|\n|-|-|\n%s"
370            % (
371                "\n".join(
372                    [f"|{key}|{value}|" for key, value in vars(self.args).items()]
373                )
374            ),
375        )
376
377        # Start the game
378        obs, infos = self.env.reset()
379        self.assigned_goals = None
380        done = False
381        episodic_return = np.zeros(self.num_env_agents)
382        if self.args.save_all_trace:
383            trace = Trace()
384        else:
385            trace = RewardTrace()
386        episode = 0
387        all_done = 0
388
389        for global_step in tqdm(
390            range(self.args.total_timesteps), desc="Training step:"
391        ):
392            # Reset
393            # ---------------------------
394            if all_done == self.num_env_agents:
395                obs, infos = self.env.reset()
396                self.assigned_goals = None  # Reset the agents goals after each reset
397                all_done = 0
398                episodic_return = np.zeros(self.num_env_agents)
399                trace.empty()
400                episode += 1
401            # ---------------------------
402
403            # Get actions
404            # ---------------------------
405            actions = self.act(observation=obs)
406            # ---------------------------
407
408            # Perform action
409            # ---------------------------
410            next_obs, reward, terminated, truncated, info = self.env.step(actions)
411            # ---------------------------
412
413            # Save data and log
414            # ---------------------------
415            trace.add(
416                actions=actions,
417                observations=obs,
418                rewards=reward,
419                terminations=terminated,
420                truncations=truncated,
421                infos=info,
422                episode=episode,
423            )
424
425            all_done = 0
426            for agent_idx, agent_name in enumerate(obs):
427                if terminated[agent_name] or truncated[agent_name]:
428                    done = True
429                    all_done += 1
430                else:
431                    done = False
432                episodic_return[agent_idx] += reward[agent_name]
433                if done:
434                    if self.args.verbose:
435                        print(
436                            f"global_step={global_step}, Agent: {agent_name} - Ep. return={episodic_return[agent_idx]}"
437                        )
438                    writer.add_scalar(
439                        f"returns/{agent_name}",
440                        episodic_return[agent_idx],
441                        global_step,
442                    )
443
444            if all_done == self.num_env_agents:
445                writer.add_scalar(
446                    "returns/total",
447                    np.sum(episodic_return),
448                    global_step,
449                )
450                if self.args.verbose:
451                    print(
452                        f"global_step={global_step} - Total ep return: {np.sum(episodic_return)}"
453                    )
454                if hasattr(trace, "add_final_obs"):
455                    trace.add_final_obs(next_obs)
456                trace.save_trace(save_path=save_path, episode=episode)
457            # ---------------------------
458
459            obs = next_obs
460
461        writer.close()
462        self.trained = True

464    def act(self, observation: Dict[str, np.ndarray]) -> dict:
465        """
466        Execute the action for each agent based on their observations and assigned goals.
467
468        This method assigns goals to agents if they haven't been assigned yet, and then
469        gets actions from each low-level agent based on their individual observations.
470
471        Args:
472            observation (Dict[str, np.ndarray]): Dictionary mapping agent names to their
473                observations as numpy arrays.
474
475        Returns:
476            dict: Dictionary mapping agent names to their chosen actions.
477        """
478        # Update goals for subagents
479        if self.assigned_goals is None:
480            self.assigned_goals = self.goal_assigner.assign_goals(observation)
481        for agent, goal in zip(self.LL_agents, self.assigned_goals):
482            agent.give_goal(goal=self.assigned_goals[goal])
483
484        actions = {}
485        for agent_name, ll_agent in zip(observation, self.LL_agents):
486            actions[agent_name] = ll_agent.act(observation=observation[agent_name])
487        return actions

191        between the agent's current position and its goal, prioritizing movement

193
