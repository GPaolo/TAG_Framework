 15    """A class for plotting and analyzing reward data from experiments.
 16
 17    This class provides functionality to load, process, and visualize reward data from

 19    with options for smoothing, confidence intervals, and standard deviation visualization.

 37        plotter = Plotter("path/to/experiments", "individual")
 38        experiments = {
 39            "exp1": {"agents": 4, "recalc": False},

 49    @staticmethod
 50    def load_test_rewards(
 51        exp_folders: List[Path], trace_type: str = "mean", verbose: int = 0
 52    ) -> pd.DataFrame | None:
 53        """
 54        Load test rewards from evaluation folders and compute statistics across seed runs.
 55
 56        This method loads reward traces from evaluation folders, processes them, and computes mean
 57        and standard deviation across different random seeds.
 58
 59        Args:
 60            exp_folders (List[Path]): List of paths to experiment folders for different seeds
 61            trace_type (str): Type of trace to load (e.g. 'mean', 'total')
 62            verbose (int, optional): Verbosity level for error reporting. Defaults to 0.
 63
 64        Returns:
 65            pd.DataFrame | None: DataFrame containing mean and std of rewards across seeds,
 66                with columns ['mean', 'std']. Returns None if no data could be loaded.
 67
 68        Notes:
 69            - Expects a folder structure with evaluations in 'eval_X' folders
 70            - Looks for CSV files named '{trace_type}_reward_reward_trace.csv'
 71            - Handles both direct placement and 'traces' subfolder for reward files
 72            - Averages rewards across multiple evaluations within each seed
 73            - Computes statistics across seeds
 74        """
 75        seed_runs = []
 76        for seed_path in exp_folders:
 77            try:
 78                evaluation = 0
 79                eval_path = seed_path / "evaluations"
 80                eval_rewards = []
 81                while (eval_path / f"eval_{evaluation}").exists():
 82                    if (eval_path / f"eval_{evaluation}" / "traces").exists():
 83                        reward_df_path = eval_path / f"eval_{evaluation}" / "traces"
 84                    else:
 85                        reward_df_path = eval_path / f"eval_{evaluation}"
 86                    reward_df_path = (
 87                        reward_df_path / f"{trace_type}_reward_reward_trace.csv"
 88                    )
 89
 90                    df = pd.read_csv(reward_df_path, index_col="episode")
 91                    eval_rewards.append(df)
 92
 93                    evaluation += 1
 94                if len(eval_rewards):
 95                    eval_rewards = pd.concat(eval_rewards, axis=1)
 96
 97                    rewards_names = list(eval_rewards.columns)
 98                    env_rewards = []
 99                    for r_name in rewards_names:
100                        if "reward_env_" in r_name:
101                            env_rewards.append(r_name)
102
103                    if not env_rewards:
104                        env_rewards = rewards_names
105
106                    # Mean reward across evaluations in the same seed
107                    seed_runs.append(eval_rewards.mean(axis=1))
108            except Exception as e:
109                if verbose >= 2:
110                    print(f"Cannot load test data for {seed_path}")
111                    print(e)
112
113        if len(seed_runs):
114            mean_reward = pd.concat(seed_runs, axis=1).mean(axis=1)
115            std_reward = pd.concat(seed_runs, axis=1).std(axis=1)
116            rewards = pd.concat([mean_reward, std_reward], axis=1)
117            rewards.rename(columns={0: "mean", 1: "std"}, inplace=True)
118            return rewards
119        else:
120            return None

122    @staticmethod
123    def load_train_rewards(
124        exp_folders: List[Path],
125        recalc: bool,
126        trace_type: str,
127        level: str | None = None,
128        verbose: int = 0,
129    ) -> pd.DataFrame | None:
130        """Load and process training rewards from experiment folders.
131
132        This method loads training rewards from multiple experiment folders, processes them,
133        and calculates statistics across seeds including mean, standard deviation and confidence intervals.
134
135        Args:
136            exp_folders (List[Path]): List of paths to experiment folders
137            recalc (bool): Whether to recalculate rewards even if cached files exist
138            trace_type (str): Type of trace to load ('train', 'test', etc.)
139            level (str | None, optional): Name of the level from where to load rewards from. Defaults to None.
140            verbose (int, optional): Verbosity level for printing progress. Defaults to 0.
141
142        Returns:
143            pd.DataFrame | None: DataFrame containing reward statistics (mean, std, confidence intervals)
144                                across seeds, or None if no valid rewards found.
145                                Columns are ['mean', 'std', 'cil', 'ciu'] where:
146                                - mean: Mean reward across seeds
147                                - std: Standard deviation across seeds
148                                - cil: Lower 95% confidence interval
149                                - ciu: Upper 95% confidence interval
150
151        Raises:
152            FileNotFoundError: If interface_level.json is not found when level='env'.
153            interface_level.json contains the name of the level of the hierarchy that interfaces with the env.
154            It should be save by the agent and is structured like:
155            {
156                "name": "bottom"
157            }
158        """
159        seed_runs = []
160        lenghts = {}
161        for seed_path in exp_folders:
162            # Make path where to look for
163            rew_df_path = seed_path / "training"
164            if level is None:
165                if (seed_path / "training" / "traces").exists():
166                    rew_df_path = seed_path / "training" / "traces"
167                else:
168                    rew_df_path = seed_path / "training"  # For old exps
169            elif level == "env":  # The true rewards are from the interface level
170                if (seed_path / "interface_level.json").exists():
171                    interface_level_name = json.load(
172                        open(seed_path / "interface_level.json", "r")
173                    )["name"]
174                else:
175                    raise FileNotFoundError(f"No interface_level.json at {seed_path}")
176                rew_df_path = seed_path / "training" / interface_level_name / "traces"
177            else:
178                rew_df_path = seed_path / "training" / level / "traces"
179
180            rew_df_path = rew_df_path / f"{trace_type}_reward.csv"
181            if rew_df_path.exists() and not recalc:
182                df = pd.read_csv(rew_df_path, index_col="episode")
183            else:
184                try:
185                    df = get_concat_traces(
186                        data_path=rew_df_path.parent,
187                        file_basename=f"{trace_type}_reward_trace_ep",
188                    )
189                except ValueError:
190                    try:
191                        df = get_concat_traces(
192                            data_path=rew_df_path.parent,
193                            file_basename=f"{trace_type}_reward_reward_trace_ep",
194                        )
195                    except ValueError:
196                        continue
197
198                df.to_csv(rew_df_path)
199
200            try:
201                params = json.load(open(seed_path / "params.json"))
202            except Exception as error:
203                params = None
204                print(f"Error loading params {seed_path}")
205                print(error)
206
207            if params is not None:
208                total_ts = params.get("total_timesteps", np.inf)
209                ep_length = params.get("episode_length", np.inf)
210                if len(df) * ep_length < total_ts:
211                    lenghts[seed_path] = f"{len(df)*ep_length}/{total_ts}"
212                # pass
213
214            rewards_names = list(df.columns)
215            env_rewards = []
216            for r_name in rewards_names:
217                if "reward_env_" in r_name:
218                    env_rewards.append(r_name)
219
220            if not env_rewards:
221                env_rewards = rewards_names
222
223            seed_runs.append(df[env_rewards].sum(axis=1))
224            if verbose >= 0:
225                print("#", end="")
226
227        if verbose >= 2:
228            if len(lenghts):
229                print()
230            for seed, ts_done in lenghts.items():
231                print(f"Seed: {seed} - {ts_done}")
232
233        n = len(seed_runs)
234        if n:
235            mean_reward = pd.concat(seed_runs, axis=1).mean(axis=1)
236            std_reward = pd.concat(seed_runs, axis=1).std(axis=1)
237
238            standard_error = std_reward / np.sqrt(n)
239            t_value = stats.t.ppf(
240                0.975, df=n - 1
241            )  # 0.975 because it's two-tailed (0.95 + 0.025)
242            # Calculate confidence intervals
243            ci_lower = mean_reward - t_value * standard_error
244            ci_upper = mean_reward + t_value * standard_error
245
246            rewards = pd.concat([mean_reward, std_reward, ci_lower, ci_upper], axis=1)
247            rewards.rename(
248                columns={0: "mean", 1: "std", 2: "cil", 3: "ciu"}, inplace=True
249            )
250            return rewards
251        else:
252            return None

254    def load_rewards(
255        self,
256        experiments: dict,
257        seeds: str | list,
258        verbose: int = 0,
259        load_test: bool = False,
260        scale_by_agents: bool = True,
261    ) -> Tuple[dict, dict]:
262        """Load reward data from experiments.
263
264        This method loads training and optionally test rewards from multiple experiments with different seeds.
265
266        Args:
267            experiments (dict): Dictionary containing experiment configurations. Each experiment should have:
268                - 'agents': Number of agents in experiment
269                - 'recalc': Boolean flag for recalculation
270                - 'level': (Optional) Level specification
271                - 'name': (Optional) Display name for the experiment
272            seeds (Union[str, list]): Seeds to load, can be either a string pattern or list of specific seeds
273            verbose (int, optional): Verbosity level for output. Defaults to 0.
274            load_test (bool, optional): Whether to load test rewards. Defaults to False.
275            scale_by_agents (bool, optional): Whether to scale rewards by number of agents. Defaults to True.
276
277        Returns:
278            Tuple[dict, dict]: A tuple containing:
279                - training_rewards (dict): Dictionary mapping experiment names to training reward data
280                - test_rewards (dict): Dictionary mapping experiment names to test reward data, or None if load_test=False
281
282        Examples:
283            >>> plotter = RewardPlotter()
284            >>> experiments = {
285            ...     "exp1": {"agents": 2, "recalc": False},
286            ...     "exp2": {"agents": 3, "recalc": True}
287            ... }
288            >>> train_rewards, test_rewards = plotter.load_rewards(experiments, seeds="*")
289        """
290        # Find experiment folders depending on the seeds
291        experiment_folders = {}
292        for exp in experiments:
293            exp_name = f"{exp}__a{experiments[exp]['agents']}"
294            experiment_folders[exp] = find_seeds(
295                exp_path=self.exp_path, exp_name=exp_name, seeds=seeds
296            )
297
298        # Get experiment data
299        training_rewards = {}
300        test_rewards = {}
301        for exp in experiment_folders:
302            if verbose >= 0:
303                print(f"Loading {exp} - {len(experiment_folders[exp])} seeds: ", end="")
304            exp_data = self.load_train_rewards(
305                experiment_folders[exp],
306                recalc=experiments[exp]["recalc"],
307                trace_type=self.trace_type,
308                level=experiments[exp].get("level", None),
309                verbose=verbose,
310            )
311            if verbose >= 0:
312                print("")
313
314            exp_name = experiments[exp].get("name", None)
315            if exp_name is None:
316                exp_name = exp
317
318            if exp_data is not None:
319                # Scale by the number of agents
320                if scale_by_agents:
321                    training_rewards[exp_name] = exp_data / experiments[exp]["agents"]
322                else:
323                    training_rewards[exp_name] = exp_data
324            else:
325                print(f"No train data for {exp}")
326
327            if load_test:
328                test_data = self.load_test_rewards(
329                    exp_folders=experiment_folders[exp],
330                    trace_type=self.trace_type,
331                    verbose=verbose,
332                )
333                # Scale by the number of agents
334                if test_data is not None:
335                    if scale_by_agents:
336                        test_rewards[exp_name] = test_data / experiments[exp]["agents"]  # type: ignore
337                    else:
338                        test_rewards[exp_name] = test_data / experiments[exp]["agents"]  # type: ignore
339                else:
340                    print(f"No test data for {exp}")
341            else:
342                test_rewards = None
343
344        return training_rewards, test_rewards  # type: ignore

346    def _plot_rewards_training(
347        self,
348        experiment_data: dict,
349        oracle_data: dict,
350        smoothing_window: int = 101,
351        poly_order: int = 3,
352        plot_std: bool = False,
353        plot_ci: bool = False,
354        max_x: None | int = None,
355        figsize: tuple = (12, 6),
356        figure: Tuple[Figure, Axes] | None = None,
357        style="classic",
358    ) -> Figure:
359        """
360        Plot training rewards using matplotlib.
361
362        Args:
363            experiment_data: Dictionary containing experiment data with 'mean', 'std', 'cil', 'ciu' columns
364            oracle_data: Dictionary containing oracle data
365            smoothing_window: Window size for Savitzky-Golay filter
366            poly_order: Polynomial order for Savitzky-Golay filter
367            plot_std: Whether to plot standard deviation bands
368            plot_ci: Whether to plot confidence interval bands
369            max_x: Maximum number of episodes to plot
370            figsize: Figure size tuple (width, height)
371
372        Returns:
373            matplotlib.figure.Figure: The created figure
374        """
375        if plot_ci and plot_std:
376            raise ValueError("Can plot only one between CI and STD")
377
378        if smoothing_window % 2 == 0:
379            print("Smoothing window size is even. Making it odd.")
380            smoothing_window += 1
381
382        # Find the maximum number of episodes
383        max_episodes = 0
384        episodes = None
385        for exp in experiment_data:
386            if len(experiment_data[exp].index) > max_episodes:
387                episodes = experiment_data[exp].index
388                max_episodes = len(episodes)
389
390        plt.style.use(style)
391        plt.rcParams.update(
392            {
393                "font.size": 14,  # Base font size * scale
394                "axes.titlesize": 14 * 1.5,
395                "axes.labelsize": 12 * 1.5,
396                "xtick.labelsize": 10 * 1.5,
397                "ytick.labelsize": 10 * 1.5,
398                "legend.fontsize": 10 * 1.5,
399                # "axes.spines.top": False,
400                # "axes.spines.right": False,
401            }
402        )
403
404        # Create figure
405        if figure is None:
406            fig, ax = plt.subplots(figsize=figsize)
407        else:
408            fig, ax = figure
409
410        # Set up color cycle (similar to Bokeh's Category20)
411        colors = plt.cm.Set1(np.linspace(0, 1, len(experiment_data)))  # type: ignore
412        # colors = plt.cm.viridis(np.linspace(0, 0.8, max(len(experiment_data), 3)))
413
414        # Plot oracle reference lines
415        for oracle_name in oracle_data:
416            ax.axhline(
417                y=oracle_data[oracle_name].mean()["mean"],
418                color="red",
419                linestyle=":",
420                alpha=1,
421            )
422
423        # Plot each experiment
424        for i, exp in enumerate(experiment_data):
425            exp_data = experiment_data[exp]
426            valid_episodes = exp_data.index
427            if max_x is not None:
428                valid_episodes = valid_episodes[:max_x]
429
430            # Apply smoothing
431            mean_smoothed = savgol_filter(
432                exp_data["mean"], smoothing_window, poly_order
433            )
434            std_smoothed = savgol_filter(exp_data["std"], smoothing_window, poly_order)
435            if plot_ci:
436                cil_smoothed = savgol_filter(
437                    exp_data["cil"], smoothing_window, poly_order
438                )
439                ciu_smoothed = savgol_filter(
440                    exp_data["ciu"], smoothing_window, poly_order
441                )
442
443            # Plot mean line
444            line = ax.plot(
445                valid_episodes,
446                mean_smoothed[: len(valid_episodes)],
447                label=exp,
448                color=colors[i],
449                linewidth=1,
450                alpha=1,
451            )
452
453            # Plot standard deviation bands
454            if plot_std:
455                ax.fill_between(
456                    valid_episodes,
457                    mean_smoothed[: len(valid_episodes)]
458                    + std_smoothed[: len(valid_episodes)],
459                    mean_smoothed[: len(valid_episodes)]
460                    - std_smoothed[: len(valid_episodes)],
461                    color=colors[i],
462                    alpha=0.1,
463                    linewidth=0,
464                )
465
466            # Plot confidence interval bands
467            if plot_ci:
468                ax.fill_between(
469                    valid_episodes,
470                    ciu_smoothed[: len(valid_episodes)],  # type: ignore
471                    cil_smoothed[: len(valid_episodes)],  # type: ignore
472                    color=colors[i],
473                    alpha=0.1,
474                    linewidth=0,
475                )
476
477        # Customize plot
478        ax.set_title(f"Mean Reward by Episode")
479        ax.set_xlabel("Episode")
480        ax.set_ylabel("Reward")
481        ax.legend(title="Experiments", loc="lower right")
482        ax.grid(True, alpha=0.3)
483        # ax.set_ylim(bottom=-14)  # Set lower limit to 0
484
485        # Enable interactive data cursor
486        def on_hover(event):
487            if event.inaxes == ax:
488                cont = ax.contains(event)
489                if cont[0]:
490                    for line in ax.get_lines():
491                        if line in cont[1]["artist"]:
492                            x, y = line.get_data()
493                            ind = cont[1]["ind"][0]
494                            x_arr = np.asarray(x)
495                            y_arr = np.asarray(y)
496                            ax.set_title(f"Episode: {x_arr[ind]:.0f}, Reward: {y_arr[ind]:.2f}")
497                            fig.canvas.draw_idle()
498
499        fig.canvas.mpl_connect("motion_notify_event", on_hover)
500
501        plt.tight_layout()
502        return fig

506        experiment_data: dict,

511        """
512        Create a matplotlib visualization of rewards during testing phase.
513
514        Args:
515            experiment_data (dict): Dictionary containing experiment data where keys are experiment names

518                If provided, adds a horizontal reference line to the plot.

525                - Optional horizontal reference line if oracle_data is provided
526                - Customized axes with rotated experiment names
527        """
528
529        # Here we select a list of colors; extend or change as needed.
530        colors_palette = colors = plt.cm.Set1(np.linspace(0, 1, len(experiment_data)))  # type: ignore
531        num_experiments = max(len(experiment_data), 3)

534        plot_data = {
535            "experiment": [],
536            "mean": [],
537            "std": [],
538            "index": [],
539            "color": [],
540        }

543            try:
544                mean_val = df["mean"].iloc[0]
545                std_val = df["std"].iloc[0]
546                plot_data["mean"].append(mean_val)
547                plot_data["std"].append(std_val)
548                plot_data["experiment"].append(exp)
549                plot_data["index"].append(i)
550                plot_data["color"].append(colors[i])
551            except Exception:
552                pass
553            i += 1
554
555        # Create the figure and axis with approximate pixel dimensions converted to inches (DPI=100)
556        if figure is None:
557            fig, ax = plt.subplots(figsize=figsize)
558        else:
559            fig, ax = figure
560
561        # Plot error bars including whiskers (capsize adds whiskers)
562        ax.errorbar(
563            plot_data["index"],
564            plot_data["mean"],
565            yerr=plot_data["std"],
566            fmt="o",
567            color="black",
568            ecolor="gray",
569            capsize=5,
570            markersize=6,
571            linestyle="none",
572        )
573
574        # Optionally, plot each point with its associated color (if desired)
575        for idx, mean_val, col in zip(
576            plot_data["index"], plot_data["mean"], plot_data["color"]
577        ):
578            ax.plot(idx, mean_val, "o", color=col, markersize=8)
579
580        # Add horizontal reference line if oracle_data provided
581        if oracle_data is not None:
582            reference_value = next(iter(oracle_data.values()))["mean"].iloc[0]
583            ax.axhline(
584                y=reference_value,
585                color="red",
586                linestyle="--",
587                linewidth=2,
588            )
589
590        # Customize axes
591        ax.set_title("Mean reward Test")
592        ax.set_xlabel("Algorithm")
593        ax.set_ylabel("Mean Value")
594        ax.set_xticks(plot_data["index"])
595        ax.set_xticklabels(plot_data["experiment"], rotation=45)
596
597        plt.tight_layout()
598        return fig

600    def plot(
601        self,
602        experiments: dict,
603        seeds: str | list,
604        smoothing_window: int = 101,
605        poly_order: int = 3,
606        notebook: bool = True,  # Ignored for matplotlib
607        plot_std: bool = False,
608        plot_ci: bool = False,
609        verbose: int = 0,
610        plot_test: bool = False,
611        scale_by_agents: bool = True,
612        max_x: int | None = None,
613        style: str = "classic",
614    ):
615        """
616        Plot training (and testing) rewards for multiple experiments using matplotlib.
617
618        The method creates plots comparing different experiments, with special handling for "oracle" experiments
619        that are plotted separately. It can show both training curves and test results side by side.
620
621        Args:
622            experiments (dict): Dictionary containing experiment data to plot.
623            seeds (str | list): Seeds to load data from, can be a string pattern or list of specific seeds.
624            smoothing_window (int, optional): Window size for smoothing the curves. Defaults to 101.
625            poly_order (int, optional): Polynomial order for smoothing. Defaults to 3.
626            notebook (bool, optional): (Ignored) For compatibility.
627            plot_std (bool, optional): Whether to plot standard deviation. Defaults to False.
628            plot_ci (bool, optional): Whether to plot confidence intervals. Defaults to False.
629            verbose (int, optional): Verbosity level for logging. Defaults to 0.
630            plot_test (bool, optional): Whether to include test results plot. Defaults to False.
631            scale_by_agents (bool, optional): Whether to scale rewards by number of agents. Defaults to True.
632            max_x (int | None, optional): Maximum x-axis value. Defaults to None.
633            style (str, optional): Plot style to use. Defaults to "classic".
634
635        Returns:
636            matplotlib.figure.Figure: The generated matplotlib figure.
637        """
638        if plot_ci and plot_std:
639            raise ValueError("Can plot only one between CI and STD")
640
641        oracle_experiments = {}

 56
 57        This method loads reward traces from evaluation folders, processes them, and computes mean
 58        and standard deviation across different random seeds.
 59
 60        Args:
 61            exp_folders (List[Path]): List of paths to experiment folders for different seeds
 62            trace_type (str): Type of trace to load (e.g. 'mean', 'total')
 63            verbose (int, optional): Verbosity level for error reporting. Defaults to 0.
 64
 65        Returns:
 66            pd.DataFrame | None: DataFrame containing mean and std of rewards across seeds,
 67                with columns ['mean', 'std']. Returns None if no data could be loaded.
 68
 69        Notes:
 70            - Expects a folder structure with evaluations in 'eval_X' folders
 71            - Looks for CSV files named '{trace_type}_reward_reward_trace.csv'
 72            - Handles both direct placement and 'traces' subfolder for reward files
 73            - Averages rewards across multiple evaluations within each seed
 74            - Computes statistics across seeds
 75        """
 76        seed_runs = []
 77        for seed_path in exp_folders:
 78            try:
 79                evaluation = 0
 80                eval_path = seed_path / "evaluations"
 81                eval_rewards = []
 82                while (eval_path / f"eval_{evaluation}").exists():
 83                    if (eval_path / f"eval_{evaluation}" / "traces").exists():
 84                        reward_df_path = eval_path / f"eval_{evaluation}" / "traces"
 85                    else:
 86                        reward_df_path = eval_path / f"eval_{evaluation}"
 87                    reward_df_path = (
 88                        reward_df_path / f"{trace_type}_reward_reward_trace.csv"
 89                    )
 90
 91                    df = pd.read_csv(reward_df_path, index_col="episode")
 92                    eval_rewards.append(df)
 93
 94                    evaluation += 1
 95                if len(eval_rewards):
 96                    eval_rewards = pd.concat(eval_rewards, axis=1)
 97
 98                    rewards_names = list(eval_rewards.columns)
 99                    env_rewards = []
100                    for r_name in rewards_names:
101                        if "reward_env_" in r_name:
102                            env_rewards.append(r_name)
103
104                    if not env_rewards:
105                        env_rewards = rewards_names
106
107                    # Mean reward across evaluations in the same seed
108                    seed_runs.append(eval_rewards.mean(axis=1))
109            except Exception as e:
110                if verbose >= 2:
111                    print(f"Cannot load test data for {seed_path}")
112                    print(e)
113
114        if len(seed_runs):
115            mean_reward = pd.concat(seed_runs, axis=1).mean(axis=1)
116            std_reward = pd.concat(seed_runs, axis=1).std(axis=1)
117            rewards = pd.concat([mean_reward, std_reward], axis=1)
118            rewards.rename(columns={0: "mean", 1: "std"}, inplace=True)
119            return rewards
120        else:
121            return None

125        exp_folders: List[Path],
126        recalc: bool,
127        trace_type: str,
128        level: str | None = None,
129        verbose: int = 0,
130    ) -> pd.DataFrame | None:
131        """Load and process training rewards from experiment folders.
132
133        This method loads training rewards from multiple experiment folders, processes them,
134        and calculates statistics across seeds including mean, standard deviation and confidence intervals.
135
136        Args:
137            exp_folders (List[Path]): List of paths to experiment folders
138            recalc (bool): Whether to recalculate rewards even if cached files exist
139            trace_type (str): Type of trace to load ('train', 'test', etc.)
140            level (str | None, optional): Name of the level from where to load rewards from. Defaults to None.
141            verbose (int, optional): Verbosity level for printing progress. Defaults to 0.
142
143        Returns:
144            pd.DataFrame | None: DataFrame containing reward statistics (mean, std, confidence intervals)
145                                across seeds, or None if no valid rewards found.
146                                Columns are ['mean', 'std', 'cil', 'ciu'] where:
147                                - mean: Mean reward across seeds
148                                - std: Standard deviation across seeds
149                                - cil: Lower 95% confidence interval
150                                - ciu: Upper 95% confidence interval
151
152        Raises:
153            FileNotFoundError: If interface_level.json is not found when level='env'.
154            interface_level.json contains the name of the level of the hierarchy that interfaces with the env.
155            It should be save by the agent and is structured like:
156            {
157                "name": "bottom"
158            }
159        """
160        seed_runs = []
161        lenghts = {}
162        for seed_path in exp_folders:
163            # Make path where to look for
164            rew_df_path = seed_path / "training"
165            if level is None:
166                if (seed_path / "training" / "traces").exists():
167                    rew_df_path = seed_path / "training" / "traces"
168                else:
169                    rew_df_path = seed_path / "training"  # For old exps
170            elif level == "env":  # The true rewards are from the interface level
171                if (seed_path / "interface_level.json").exists():
172                    interface_level_name = json.load(
173                        open(seed_path / "interface_level.json", "r")
174                    )["name"]
175                else:
176                    raise FileNotFoundError(f"No interface_level.json at {seed_path}")
177                rew_df_path = seed_path / "training" / interface_level_name / "traces"
178            else:
179                rew_df_path = seed_path / "training" / level / "traces"
180
181            rew_df_path = rew_df_path / f"{trace_type}_reward.csv"
182            if rew_df_path.exists() and not recalc:
183                df = pd.read_csv(rew_df_path, index_col="episode")
184            else:
185                try:
186                    df = get_concat_traces(
187                        data_path=rew_df_path.parent,
188                        file_basename=f"{trace_type}_reward_trace_ep",
189                    )
190                except ValueError:
191                    try:
192                        df = get_concat_traces(
193                            data_path=rew_df_path.parent,
194                            file_basename=f"{trace_type}_reward_reward_trace_ep",
195                        )
196                    except ValueError:
197                        continue
198
199                df.to_csv(rew_df_path)
200
201            try:
202                params = json.load(open(seed_path / "params.json"))
203            except Exception as error:
204                params = None
205                print(f"Error loading params {seed_path}")
206                print(error)
207
208            if params is not None:
209                total_ts = params.get("total_timesteps", np.inf)
210                ep_length = params.get("episode_length", np.inf)
211                if len(df) * ep_length < total_ts:
212                    lenghts[seed_path] = f"{len(df)*ep_length}/{total_ts}"
213                # pass
214
215            rewards_names = list(df.columns)
216            env_rewards = []
217            for r_name in rewards_names:
218                if "reward_env_" in r_name:
219                    env_rewards.append(r_name)
220
221            if not env_rewards:
222                env_rewards = rewards_names
223
224            seed_runs.append(df[env_rewards].sum(axis=1))
225            if verbose >= 0:
226                print("#", end="")
227
228        if verbose >= 2:
229            if len(lenghts):
230                print()
231            for seed, ts_done in lenghts.items():
232                print(f"Seed: {seed} - {ts_done}")
233
234        n = len(seed_runs)
235        if n:
236            mean_reward = pd.concat(seed_runs, axis=1).mean(axis=1)
237            std_reward = pd.concat(seed_runs, axis=1).std(axis=1)
238
239            standard_error = std_reward / np.sqrt(n)
240            t_value = stats.t.ppf(
241                0.975, df=n - 1
242            )  # 0.975 because it's two-tailed (0.95 + 0.025)
243            # Calculate confidence intervals
244            ci_lower = mean_reward - t_value * standard_error
245            ci_upper = mean_reward + t_value * standard_error
246
247            rewards = pd.concat([mean_reward, std_reward, ci_lower, ci_upper], axis=1)
248            rewards.rename(
249                columns={0: "mean", 1: "std", 2: "cil", 3: "ciu"}, inplace=True
250            )
251            return rewards
252        else:
253            return None
